            recoveryOriginPhase = null;
            recoveryWaits = 0;

            return;

            return;

            log("[RECOVERY ABORT] Budget ciclo 2% esaurito");
            recoveryStep = 0;
            recoveryOriginPhase = null;
            recoveryWaits = 0;
            closeCycle();
            return;
        }

        if (!canBet(recBet)) return;

        lastMode = recoveryStep === 1 ? "RECOVERY_1" : "RECOVERY_2";
        placeBet(recBet, recPayout, lastMode, snapshot, confidence, targetPotential, pace);
        return;
    }

    /* Minimum spacing: frozen in NORMAL; adaptive/free in Recovery Plan. */
    var minSpacing;
    if (recoveryPlanActive) {
        minSpacing = lastRecoveryPlanSignal >= 88 ? 0 : (pace === "SLOW" ? 2 : 1);
    } else {
        minSpacing = pace === "SLOW" ? 3 : (pace === "NORMAL" ? 2 : 1);
    }
    if (roundsSinceBet < minSpacing) {
        logDecision("SPACING", snapshot, confidence, targetPotential, pace, volatilityPenalty, roundsSinceBet + "/" + minSpacing);
        return;
    }

    /* Entry decision: Sentinel HUNT competes for the ONE bet, not an extra bet.
       In normal configuration HUNT runs only in virtual parallel. */
    var entry = evaluateEntry(snapshot, confidence, targetPotential, volatilityPenalty, pace, attackBlocked);
    if (entry.enter && entry.mode === "ATTACK" && ksiGuardAttack()) {
        KSI_EXPLAIN.lastAttack = "ATTACK ridotto a BASE: ROI rolling negativo";
        entry.mode = "BASE";
        entry.payout = chooseAdaptivePayout(snapshot, confidence, targetPotential, pace, false, false);
        entry.amount = chooseAdaptiveBetSize(snapshot, confidence, targetPotential, pace, false, false);
    }
    var huntCandidate = ksiChooseHunt(snapshot);
    if (huntCandidate && ksiHuntMayGoLive(huntCandidate) && !recoveryPlanActive
        && !attackBlocked && entry.enter) {
        entry = {enter:true,mode:"HUNT",amount:unit,payout:huntCandidate.target,
                 reason:"SENTINEL HUNT VALIDATED " + huntCandidate.score.toFixed(1)};
    }

    // V2.2: selezione basata su redditività rolling, non solo % vittorie.
    entry = k22ReviewEntry(entry, snapshot, confidence, targetPotential);
    entry = k24Supervise(entry,snapshot,confidence,targetPotential,pace,attackBlocked);
    // V2.7: tutte le famiglie sono candidate, ma nessuna bypassa il controllo del capitale.
    if(config.enableEnsemble.value)entry=K27_ENSEMBLE.choose(entry,K22_STATE,{attackBlocked:attackBlocked,phase:currentPhase,confidence:confidence,targetPotential:targetPotential,givebackBrake:givebackBrake});
    K30_DRIFT.plan(entry);
    entry=K27_ENSEMBLE.size(entry,runtimeBalance(),K22_STATE);
    // V2.6 Governor continua a poter sospendere scelte con esiti negativi.
    entry = K26_GOVERNOR.gate(entry, K24_LAST_PLAN);
    entry = K30_DRIFT.gate(entry, K22_STATE);
    if (!entry.enter) {
        K27_ENSEMBLE.noteVeto('AFTER_SELECTION: ' + String(entry.reason||'UNKNOWN'));
        logDecision("SKIP", snapshot, confidence, targetPotential, pace, volatilityPenalty, entry.reason);
        return;
    }

    startCycleIfNeeded();
    if (!k22AllowMoney(entry.amount,entry.mode)) {
        K27_ENSEMBLE.noteVeto('MONEY_GATE');return;
    }

    if (!canUseCycleBudget(entry.amount)) {
        K27_ENSEMBLE.noteVeto('CYCLE_BUDGET');
        log("[SKIP] Cycle budget limit | amount", bits(entry.amount), "budget", bits(cycleBudget));
        closeCycle();
        return;
    }

    if (!canBet(entry.amount)) {
        K27_ENSEMBLE.noteVeto('INSUFFICIENT_CAPITAL');return;
    }

    lastMode = entry.mode;

    if (entry.mode === "ATTACK") {
        attackLock = currentPhase === "HOT" ? 3 : 5;
    }

    placeBet(entry.amount, entry.payout, entry.mode, snapshot, confidence, targetPotential, pace);
    if(!playedThisRound)K27_ENSEMBLE.noteVeto('BET_NOT_PLACED');
});

engine.on("GAME_ENDED", function () {
    if (stopped) return;

    var game = engine.history.first();
    if (!game || typeof game.bust !== "number") return;
    if (!STRIKER_LIVE && shadowCorePending) {
        var shadowWon = game.bust >= shadowCorePending.payout;
        var shadowPL = shadowWon ? Math.round(shadowCorePending.amount * (shadowCorePending.payout - 1)) : -shadowCorePending.amount;
        runtimeShadowBalance += shadowPL;
        game = {bust:game.bust, wager:shadowCorePending.amount,
                cashedAt:shadowWon ? shadowCorePending.payout : 0};
        shadowCorePending = null;
    }
    /* No lookahead: score the last candidate and then train on the new outcome. */
    ksiSettleLearning(game.bust);
    // Valuta ogni proposta virtuale e gli ingressi mancati sul round appena chiuso.
    K24_BRAIN.settle(game.bust);
    K25_PATTERNS.settle(game.bust);
    K26_GOVERNOR.settle(game.bust);
    K27_CLASSIC.settle(game.bust);
    K27_DEEP.settle(game.bust);
    K27_ENSEMBLE.settle(game.bust);
    K30_DRIFT.settle(game.bust);

    totalRounds++;
    var checkpointN=Math.floor(Number(config.mlCheckpointRounds.value)||0);
    if(checkpointN>=1000 && totalRounds%checkpointN===0)k24ExportMemory();

    history.push(game.bust);
    if (history.length > MEMORY_MAX) history.shift();
    updateFiveXIntelligence(game.bust);
    // Finestra valutazione flessibile 90/100/110, non modifica il limite rolling puntate.
    k22SessionReview();

    if (trackingThisRound) {
        decisionWindow.push(playedThisRound ? 1 : 0);
        if (decisionWindow.length > 100) decisionWindow.shift();
    }

    if (!playedThisRound || !game.wager) {
        k22RoundFeedback(false,0);
        roundsSinceBet++;
        if(totalRounds % 100 === 0)printStats();
        k26MaybePersist();
        return;
    }

    roundsSinceBet = 0;
    totalBets++;

    var won = !!game.cashedAt;
    var wager = game.wager;
    var pnl = won ? wager * (game.cashedAt - 1) : -wager;

    ksiSettleCore(won, pnl, wager, game.bust);
    k22RoundFeedback(true,pnl);
    addModePnl(lastMode, pnl);
    phasePnl[lastPhase] += pnl;
    recordOutcome(won, pnl, lastMode, lastPhase, lastPayout);
    if (lastRecoveryPlanBet) {
        recoveryPlanBets++;
        pnlRecoveryPlan += pnl;
    }

    if (won) {
        wins++;
        consecutiveLosses = 0;
        logResult("WIN", game, pnl);

        recoveryStep = 0;
        recoveryOriginPhase = null;
        recoveryWaits = 0;
        closeCycle();
    } else {
        losses++;
        consecutiveLosses++;
        logResult("LOSS", game, pnl);

        if ((lastMode === "BASE" || lastMode === "ATTACK") &&
            K22_STATE !== 'OBSERVE' && K22_STATE !== 'HARD_STOP') {
            recoveryStep = 1;
            recoveryOriginPhase = lastPhase;
            recoveryWaits = 0;
        } else if (lastMode === "RECOVERY_1" && K22_STATE !== 'OBSERVE' && K22_STATE !== 'HARD_STOP') {
            recoveryStep = 2;
            recoveryWaits = 0;
        } else if (lastMode === "RECOVERY_2") {
            recoveryStep = 0;
            recoveryOriginPhase = null;
            recoveryWaits = 0;
            cooldown = Math.max(cooldown, 8);
            closeCycle();
        }

        if (consecutiveLosses >= 2 && recoveryStep === 0) {
            cooldown = Math.max(cooldown, 5);
        }
    }

    updateBestBalance(runtimeBalance());
    updateProtection(runtimeBalance());
    updateGivebackBrake(runtimeBalance());
    updateRecoveryPlanState(runtimeBalance());
    k22AfterRound();

    if (totalRounds % 20 === 0) printStats();
    k26MaybePersist();
});

/* ==========================================================
   SNAPSHOT / OBSERVER 5-14-30
   ========================================================== */
function buildSnapshot() {
    return {
        m5: metrics(5),
        m14: metrics(14),
        m30: metrics(30),
        lowStreak: currentLowStreak(),
        highStreak: currentHighStreak(),
        recentSlope: averageSlope(5, 14),
        alternation8: extremeAlternation(8),
        alternation14: extremeAlternation(14),
        phaseAge: phaseAge
    };
}

function metrics(windowSize) {
    var n = Math.min(windowSize, history.length);
    if (!n) {
        return { n: 0, avg: 0, cappedAvg: 0, median: 0, p140: 0, p200: 0, p300: 0, p500: 0, max: 0 };
    }

    var arr = history.slice(history.length - n);
    var sorted = arr.slice().sort(function (a, b) { return a - b; });
    var sum = 0;
    var cappedSum = 0;
    var ge140 = 0, ge200 = 0, ge300 = 0, ge500 = 0, max = 0;

    for (var i = 0; i < arr.length; i++) {
        var v = arr[i];
        sum += v;
        cappedSum += Math.min(v, 10);
        if (v >= 1.40) ge140++;
        if (v >= 2.00) ge200++;
        if (v >= 3.00) ge300++;
        if (v >= 5.00) ge500++;
        if (v > max) max = v;
    }

    var median;
    if (n % 2 === 1) {
        median = sorted[(n - 1) / 2];
    } else {
        median = (sorted[n / 2 - 1] + sorted[n / 2]) / 2;
    }

    return {
        n: n,
        avg: sum / n,
        cappedAvg: cappedSum / n,
        median: median,
        p140: ge140 / n,
        p200: ge200 / n,
        p300: ge300 / n,
        p500: ge500 / n,
        max: max
    };
}

/* ==========================================================
   PHASE ENGINE
   Main rule requested by user:
   NEGATIVE < 2.00 avg14
   POSITIVE 2.00 - 3.49
   HOT >= 3.50
   with median/volatility confirmation
   ========================================================== */
function detectPhase(s) {
    var avg14 = s.m14.cappedAvg;
    var med14 = s.m14.median;

    if (avg14 < 2.00) return "NEGATIVE";

    if (avg14 >= 3.50) {
        /* Avoid fake HOT caused by one absurd spike */
        var hotSupport = 0;
        if (med14 >= 1.75) hotSupport++;
        if (s.m14.p200 >= 0.35) hotSupport++;
        if (s.m14.p300 >= 0.18) hotSupport++;
        if (s.m5.cappedAvg >= 2.20) hotSupport++;

        if (hotSupport >= 2) return "HOT";
        return "POSITIVE";
    }

    return "POSITIVE";
}

/* ==========================================================
   PHASE HYSTERESIS / HOT PERSISTENCE
   HOT must be confirmed twice before becoming operational.
   Leaving HOT also needs 2 POSITIVE observations, while
   NEGATIVE remains immediate for safety.
   ========================================================== */
function resolvePhaseWithPersistence(candidate) {
    hotTransitionCaution = false;

    if (candidate === "NEGATIVE") {
        hotCandidateStreak = 0;
        hotExitStreak = 0;
        return "NEGATIVE";
    }

    if (currentPhase === "HOT") {
        if (candidate === "HOT") {
            hotExitStreak = 0;
            hotCandidateStreak = 0;
            return "HOT";
        }

        hotExitStreak++;
        hotTransitionCaution = true;

        if (hotExitStreak >= HOT_EXIT_CONFIRM_ROUNDS) {
            hotExitStreak = 0;
            hotCandidateStreak = 0;
            return "POSITIVE";
        }

        return "HOT";
    }

    if (candidate === "HOT") {
        hotCandidateStreak++;
        hotExitStreak = 0;

        if (hotCandidateStreak >= HOT_CONFIRM_ROUNDS) {
            hotCandidateStreak = 0;
            hotPhaseBetCount = 0;
            return "HOT";
        }

        return "POSITIVE";
    }

    hotCandidateStreak = 0;
    hotExitStreak = 0;
    return "POSITIVE";
}

/* ==========================================================
   BREAK WATCH - NO AUTO BET
   ========================================================== */
function updateBreakWatch(s) {
    var info = detectNegativeBreak();

    if (info.detected) {
        breakWatchRounds = 3;
        breakWatchStrength = info.strength;
        lastBreakMultiplier = info.multiplier;
        log("[BREAK WATCH] x" + info.multiplier.toFixed(2), "| block", info.blockLength, "| strength", info.strength.toFixed(0));
        return;
    }

    if (breakWatchRounds > 0) {
        breakWatchRounds--;
        if (breakWatchRounds === 0) {
            breakWatchStrength = 0;
            lastBreakMultiplier = 0;
        }
    }
}

function detectNegativeBreak() {
    var n = history.length;
    if (n < 4) return { detected: false, multiplier: 0, blockLength: 0, strength: 0 };

    var last = history[n - 1];
    if (last < 1.40) return { detected: false, multiplier: last, blockLength: 0, strength: 0 };

    var count = 0;
    var sum = 0;

    for (var i = n - 2; i >= 0; i--) {
        if (history[i] < 1.40) {
            count++;
            sum += history[i];
        } else {
            break;
        }
    }

    if (count < 3) return { detected: false, multiplier: last, blockLength: count, strength: 0 };

    var avgBlock = sum / count;
    var strength = 35;
    strength += clamp((count - 3) * 8, 0, 24);
    strength += clamp((last - 1.40) * 12, 0, 30);
    strength += clamp((1.40 - avgBlock) * 30, 0, 11);

    return {
        detected: true,
        multiplier: last,
        blockLength: count,
        strength: clamp(strength, 0, 100)
    };
}

/* ==========================================================
   VOLATILITY PENALTY
   Light penalty, never a hard guard
   ========================================================== */
function computeVolatilityPenalty(s) {
    var penalty = 0;
    var avg = s.m14.cappedAvg;
    var med = s.m14.median;

    if (avg > med * 1.8) penalty += 12;
    if (avg > med * 2.4) penalty += 12;
    if (s.m14.max >= 20 && s.m14.p300 < 0.22) penalty += 12;
    if (s.m5.max >= 15 && s.m5.p200 < 0.40) penalty += 8;

    /* alternanza estrema tipo 1.05 -> 8x -> 1.10 -> 6x */
    if (s.alternation8 >= 0.55) penalty += 16;
    if (s.alternation8 >= 0.75) penalty += 14;
    if (s.alternation14 >= 0.45) penalty += 8;

    return clamp(penalty, 0, 55);
}

/* ==========================================================
   CONFIDENCE / PERSISTENCE 0-100
   ========================================================== */
function computeConfidence(s, volatilityPenalty) {
    var c = 45;

    c += clamp((phaseAge - 1) * 5, 0, 20);

    if (currentPhase === "NEGATIVE") {
        if (s.m14.cappedAvg < 1.80) c += 8;
        if (s.m14.median < 1.50) c += 7;
        if (s.lowStreak >= 3) c += 8;
    }

    if (currentPhase === "POSITIVE") {
        if (s.m14.median >= 1.60) c += 8;
        if (s.m14.p200 >= 0.32) c += 8;
        if (s.m5.p140 >= 0.60) c += 6;
    }

    if (currentPhase === "HOT") {
        if (s.m14.median >= 2.00) c += 10;
        if (s.m14.p300 >= 0.22) c += 10;
        if (s.m5.p200 >= 0.50) c += 8;
    }

    if (breakWatchRounds > 0) {
        c += breakWatchStrength * 0.12;
    }

    c -= volatilityPenalty;

    /* Self-calibration: recent bad decisions reduce effective confidence. */
    lastFeedback = computeOutcomeFeedback();
    c -= lastFeedback.basePenalty;

    return clamp(c, 0, 100);
}

/* ==========================================================
   OUTCOME FEEDBACK / SELF-CALIBRATION
   This does not predict future crashes. It only measures how
   well the bot's own recent decisions have been performing.
   ========================================================== */
function recordOutcome(won, pnl, mode, phase, payout) {
    outcomeMemory.push({
        won: !!won,
        pnl: pnl,
        mode: mode,
        phase: phase,
        payout: payout
    });

    if (outcomeMemory.length > OUTCOME_MEMORY_MAX) outcomeMemory.shift();
}

function selectRecentOutcomes(limit, phase, modePrefix) {
    var out = [];

    for (var i = outcomeMemory.length - 1; i >= 0 && out.length < limit; i--) {
        var item = outcomeMemory[i];
        if (phase && item.phase !== phase) continue;
        if (modePrefix && item.mode.indexOf(modePrefix) !== 0) continue;
        out.push(item);
    }

    return out;
}

function summarizeOutcomes(items) {
    var winsLocal = 0;
    var pnlLocal = 0;

    for (var i = 0; i < items.length; i++) {
        if (items[i].won) winsLocal++;
        pnlLocal += items[i].pnl;
    }

    return {
        n: items.length,
        wins: winsLocal,
        losses: items.length - winsLocal,
        winRate: items.length ? winsLocal / items.length : 0,
        pnl: pnlLocal
    };
}

function currentOutcomeLossStreak() {
    var streak = 0;

    for (var i = outcomeMemory.length - 1; i >= 0; i--) {
        if (!outcomeMemory[i].won) streak++;
        else break;
    }

    return streak;
}

function performancePenalty(summary, softLossRate, hardLossRate) {
    if (summary.n < 3 || summary.pnl >= 0) return 0;

    var p = 4;
    var lossRate = 1 - summary.winRate;

    if (lossRate >= softLossRate) p += 4;
    if (summary.n >= 4 && lossRate >= hardLossRate) p += 4;

    return p;
}

function computeOutcomeFeedback() {
    var globalSummary = summarizeOutcomes(selectRecentOutcomes(8, null, null));
    var phaseSummary = summarizeOutcomes(selectRecentOutcomes(6, currentPhase, null));
    var attackSummary = summarizeOutcomes(selectRecentOutcomes(5, null, "ATTACK"));
    var recoverySummary = summarizeOutcomes(selectRecentOutcomes(6, null, "RECOVERY"));
    var lossStreak = currentOutcomeLossStreak();

    var basePenalty = 0;
    basePenalty += performancePenalty(globalSummary, 0.55, 0.70);
    basePenalty += performancePenalty(phaseSummary, 0.55, 0.70);
    basePenalty += clamp((lossStreak - 1) * 3, 0, 9);
    basePenalty = clamp(basePenalty, 0, 24);

    var attackPenalty = performancePenalty(attackSummary, 0.50, 0.67);
    var recoveryPenalty = performancePenalty(recoverySummary, 0.50, 0.67);

    return {
        basePenalty: basePenalty,
        attackPenalty: clamp(attackPenalty, 0, 12),
        recoveryPenalty: clamp(recoveryPenalty, 0, 12),
        lossStreak: lossStreak
    };
}

/* ==========================================================
   TARGET POTENTIAL 0-100
   ========================================================== */
function computeTargetPotential(s, confidence, volatilityPenalty, balance) {
    var p = profitPct(balance);
    var nextTarget = p < 1 ? 1 : (p < 2 ? 2 : 5);
    var distance = Math.max(0, nextTarget - p);
    var roundsLeft = 999999; // continuous mode: no session horizon

    var tp = 35;
    tp += confidence * 0.35;

    if (currentPhase === "POSITIVE") tp += 8;
    if (currentPhase === "HOT") tp += 15;
    if (currentPhase === "NEGATIVE") tp -= 12;

    if (s.recentSlope > 0.30) tp += 8;
    if (s.recentSlope < -0.30) tp -= 8;

    if (distance <= 0.50) tp += 8;
    if (distance > 2.50) tp -= 8;

    if (roundsLeft < 30 && distance > 1.0) tp -= 10;

    if (protectionLevel === 1) tp -= 5;
    if (protectionLevel === 2) tp -= 12;

    var giveback = Math.max(0, bestProfitPct - p);
    if (protectionLevel === 1 && giveback > 0.40) tp -= 10;
    if (protectionLevel === 2 && giveback > 0.50) tp -= 15;
    if (protectionLevel === 2 && giveback > 1.00) tp -= 18;

    if (givebackBrake === 1) tp -= 8;
    if (givebackBrake === 2) tp -= 16;

    tp -= volatilityPenalty * 0.5;

    return clamp(tp, 0, 100);
}

/* ==========================================================
   PROFIT PROTECTION +1 / +2 / +5
   soft, progressive, not rigid
   ========================================================== */
function updateProtection(balance) {
    if (recoveryPlanActive) {
        protectionLevel = 0;
        return;
    }
    var p = profitPct(balance);
    if (p > bestProfitPct) bestProfitPct = p;

    if (bestProfitPct >= 2) {
        protectionLevel = 2;
    } else if (bestProfitPct >= 1) {
        protectionLevel = 1;
    } else {
        protectionLevel = 0;
    }
}

function updateGivebackBrake(balance) {
    if (recoveryPlanActive) {
        givebackBrake = 0;
        return;
    }
    var p = profitPct(balance);
    var giveback = Math.max(0, bestProfitPct - p);

    givebackBrake = 0;

    if (bestProfitPct >= 1.00 && giveback >= 0.35) {
        givebackBrake = 1;
    }

    if ((bestProfitPct >= 1.50 && giveback >= 0.55) ||
        (bestProfitPct >= 2.00 && giveback >= 0.50) ||
        (bestProfitPct >= 1.00 && giveback >= 0.85)) {
        givebackBrake = 2;
    }
}

/* ==========================================================
   PACE
   ========================================================== */
function choosePace(s, confidence, targetPotential, volatilityPenalty) {
    var score = confidence * 0.45 + targetPotential * 0.55;

    if (protectionLevel === 2) score -= 12;
    else if (protectionLevel === 1) score -= 5;

    var giveback = Math.max(0, bestProfitPct - profitPct(runtimeBalance()));
    if (protectionLevel === 1 && giveback > 0.40) score -= 8;
    if (protectionLevel === 2 && giveback > 0.50) score -= 14;
    if (protectionLevel === 2 && giveback > 1.00) score -= 18;

    if (givebackBrake === 1) score -= 12;
    if (givebackBrake === 2) score -= 24;
    if (hotTransitionCaution) score -= 10;

    score -= volatilityPenalty * 0.4;

    if (currentPhase === "NEGATIVE") score -= 8;
    if (currentPhase === "HOT") score += 8;

    if (givebackBrake === 2) return "SLOW";
    if (givebackBrake === 1 && score >= 72) return "NORMAL";
    if (score >= 72) return "AGGRESSIVE";
    if (score >= 50) return "NORMAL";
    return "SLOW";
}

/* ==========================================================
   ENTRY DECISION
   ========================================================== */
function evaluateEntry(s, confidence, targetPotential, volatilityPenalty, pace, attackBlocked) {
    if (recoveryPlanActive) {
        return evaluateRecoveryPlanEntry(s, confidence, targetPotential, volatilityPenalty, pace, attackBlocked);
    }

    var threshold = 0;

    if (currentPhase === "NEGATIVE") threshold = 72;
    else if (currentPhase === "POSITIVE") threshold = 58;
    else threshold = 62;

    if (pace === "SLOW") threshold += 8;
    if (pace === "AGGRESSIVE") threshold -= 6;

    if (protectionLevel === 1) threshold += 4;
    if (protectionLevel === 2) threshold += 10;

    var giveback = Math.max(0, bestProfitPct - profitPct(runtimeBalance()));
    if (protectionLevel === 1 && giveback > 0.40) threshold += 8;
    if (protectionLevel === 2 && giveback > 0.50) threshold += 14;
    if (protectionLevel === 2 && giveback > 1.00) threshold += 16;

    if (givebackBrake === 1) threshold += 6;
    if (givebackBrake === 2) threshold += 12;
    if (hotTransitionCaution) threshold += 6;

    /* Negative can play, but Break Watch only improves confidence; never auto-enters */
    if (currentPhase === "NEGATIVE" && breakWatchRounds > 0) {
        threshold -= clamp(breakWatchStrength * 0.08, 0, 6);
    }

    var decisionScore = confidence * 0.55 + targetPotential * 0.45 - volatilityPenalty * 0.25;

    if (decisionScore < threshold) {
        return { enter: false, reason: "decision " + decisionScore.toFixed(0) + " < " + threshold.toFixed(0) };
    }

    var attack = false;
    var attackConfidence = confidence - lastFeedback.attackPenalty;
    var attackMinConfidence = givebackBrake === 1 ? 82 : 72;
    var attackMinTP = givebackBrake === 1 ? 78 : 70;

    if (!attackBlocked && pace === "AGGRESSIVE" && attackConfidence >= attackMinConfidence && targetPotential >= attackMinTP) {
        if (currentPhase === "HOT" || (currentPhase === "POSITIVE" && s.m14.p200 >= 0.38)) {
            attack = true;
        }
    }

    var payout = chooseAdaptivePayout(s, confidence, targetPotential, pace, attack, false);
    var amount = chooseAdaptiveBetSize(s, confidence, targetPotential, pace, attack, false);

    return {
        enter: true,
        mode: attack ? "ATTACK" : "BASE",
        payout: payout,
        amount: amount,
        reason: "OK"
    };
}

/* ==========================================================
   ADAPTIVE PAYOUT 1.01 - 5.00
   ========================================================== */
function chooseAdaptivePayout(s, confidence, targetPotential, pace, attack, recovery) {
    if (recoveryPlanActive) {
        return chooseRecoveryPlanPayout(s, confidence, targetPotential, pace, attack, recovery);
    }

    var p;
    var med = Math.max(1.01, s.m14.median);

    if (currentPhase === "NEGATIVE") {
        p = 1.05 + confidence / 100 * 0.20;
        if (breakWatchRounds > 0) p += breakWatchStrength / 100 * 0.15;
        p = Math.min(p, Math.max(1.08, med * 0.95 + 0.10));
    } else if (currentPhase === "POSITIVE") {
        p = 1.15 + confidence / 100 * 0.35 + s.m14.p200 * 0.35 + targetPotential / 100 * 0.10;
        p = Math.min(p, Math.max(1.20, med * 0.90));
    } else {
        p = 1.35 + confidence / 100 * 0.65 + s.m14.p300 * 0.80 + targetPotential / 100 * 0.15;
        p = Math.min(p, Math.max(1.50, med * 1.05));
    }

    if (pace === "SLOW") p -= 0.10;
    if (pace === "AGGRESSIVE") p += 0.15;
    if (attack) p += 0.25;
    if (recovery) p -= 0.15;

    if (protectionLevel === 1) p -= 0.08;
    if (protectionLevel === 2) p -= 0.18;

    if (currentPhase === "HOT" && hotPhaseBetCount < 3) p -= 0.10;
    if (hotTransitionCaution) p -= 0.12;
    if (givebackBrake === 1) p -= 0.06;
    if (givebackBrake === 2) p -= 0.12;

    return roundMultiplier(clamp(p, MIN_PAYOUT, MAX_PAYOUT));
}

/* ==========================================================
   ADAPTIVE BET SIZE
   Based on fraction of 2% cycle budget
   ========================================================== */
function chooseAdaptiveBetSize(s, confidence, targetPotential, pace, attack, recovery) {
    if (recoveryPlanActive) {
        return chooseRecoveryPlanBetSize(s, confidence, targetPotential, pace, attack, recovery);
    }

    startCycleIfNeeded();

    var fraction = 0.07;

    if (currentPhase === "NEGATIVE") fraction = 0.07;
    if (currentPhase === "POSITIVE") fraction = 0.10;
    if (currentPhase === "HOT") fraction = 0.14;

    fraction += confidence / 100 * 0.06;
    fraction += targetPotential / 100 * 0.05;

    if (pace === "SLOW") fraction *= 0.65;
    if (pace === "AGGRESSIVE") fraction *= 1.15;
    if (attack) fraction *= 1.30;
    if (recovery) fraction *= recoveryStep === 2 ? 1.10 : 1.02;

    if (protectionLevel === 1) fraction *= 0.78;
    if (protectionLevel === 2) fraction *= 0.50;

    /* First 3 HOT bets are deliberately lighter. */
    if (currentPhase === "HOT" && hotPhaseBetCount < 3) fraction *= 0.72;
    if (hotTransitionCaution) fraction *= 0.75;

    if (attack && lastFeedback.attackPenalty > 0) fraction *= 0.85;
    if (recovery && lastFeedback.recoveryPenalty > 0) fraction *= 0.85;

    if (givebackBrake === 1) fraction *= 0.80;
    if (givebackBrake === 2) fraction *= 0.60;

    /* Preserve the 2.0 sizing at >=1000 bits, but automatically soften it
       for smaller balances such as 729 bits. */
    fraction *= capitalRiskScale(runtimeBalance());
    fraction = clamp(fraction, 0.03, 0.30);

    var remaining = Math.max(0, cycleBudget - cycleExposure);
    var amount = Math.floor(cycleBudget * fraction);

    amount = Math.min(amount, remaining);
    amount = Math.max(unit, amount);

    return roundSats(amount);
}

/* ==========================================================
   RECOVERY REVALIDATION
   Can continue, reduce, wait, or abort
   ========================================================== */
function evaluateRecovery(s, confidence, targetPotential, volatilityPenalty, pace) {
    var giveback = Math.max(0, bestProfitPct - profitPct(runtimeBalance()));

    if (volatilityPenalty >= 30 && confidence < 72) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "volatilità alta + confidence insufficiente" };
    }

    if (recoveryOriginPhase === "HOT" && currentPhase === "NEGATIVE") {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "fase degradata HOT -> NEGATIVE" };
    }

    if (recoveryOriginPhase === "POSITIVE" && currentPhase === "NEGATIVE") {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "fase degradata POSITIVE -> NEGATIVE" };
    }

    if (currentPhase === "NEGATIVE" && breakWatchRounds === 0 && confidence < 72) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "NEGATIVE senza Break Watch forte" };
    }

    if (protectionLevel === 2 && (targetPotential < 72 || giveback > 0.75)) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "protezione +2% / profit giveback" };
    }

    var recoveryConfidence = confidence - lastFeedback.recoveryPenalty;
    var minConf = recoveryStep === 1 ? 62 : 72;
    var minTP = recoveryStep === 1 ? 56 : 66;

    if (recoveryPlanActive) {
        var rpRecoverySignal = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
        var rpRecoveryMin = recoveryStep === 1 ? 70 : 78;
        minConf += 4;
        minTP += 4;
        if (rpRecoverySignal < rpRecoveryMin) {
            recoveryWaits++;
            if (recoveryWaits >= 2) {
                recoveryWaits = 0;
                return { action: "ABORT", reason: "Recovery Plan: segnale recovery non abbastanza forte" };
            }
            return { action: "WAIT", reason: "Recovery Plan: attende qualità recovery " + recoveryWaits + "/2" };
        }
    }

    /* Early HOT R2 requires materially stronger confirmation. */
    if (recoveryStep === 2 && recoveryOriginPhase === "HOT" && hotPhaseBetCount < 3) {
        minConf = Math.max(minConf, 88);
        minTP = Math.max(minTP, 82);
    }

    if (givebackBrake === 1) {
        minConf += 4;
        minTP += 4;
    }

    if (givebackBrake === 2 && recoveryStep === 2) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "profit giveback brake: R2 bloccato" };
    }

    if (recoveryConfidence < minConf || targetPotential < minTP) {
        recoveryWaits++;
        if (recoveryWaits >= 2) {
            recoveryWaits = 0;
            return { action: "ABORT", reason: "revalidation fallita per 2 round" };
        }
        return { action: "WAIT", reason: "attende conferma recovery " + recoveryWaits + "/2" };
    }

    recoveryWaits = 0;

    var payout = chooseAdaptivePayout(s, confidence, targetPotential, pace, false, true);
    var amount = chooseAdaptiveBetSize(s, confidence, targetPotential, pace, false, true);

    if (recoveryStep === 2) amount = roundSats(amount * 1.12);

    return { action: "BET", amount: amount, payout: payout, reason: "revalidated" };
}

/* ==========================================================
   CYCLE RISK 1.20%-1.38%
   ========================================================== */
function startCycleIfNeeded() {
    if (cycleActive) return;

    cycleActive = true;
    cycleExposure = 0;
    if (recoveryPlanActive) {
        /* Recovery is not constrained by NORMAL 2% cycle budget.
           Individual size is selected by Recovery intelligence. */
        cycleBudget = Math.max(unit, runtimeBalance());
    } else {
        var maxCyclePct = k22AllowHotBudget() ? Math.min(Number(config.hotCycleBudgetPct.value)||1.38,1.38) : 1.20;
        cycleBudget = Math.floor(runtimeBalance() * (maxCyclePct / 100));
        cycleBudget = Math.max(unit, cycleBudget);
    }
}

function canUseCycleBudget(amount) {
    startCycleIfNeeded();
    if (recoveryPlanActive) return amount > 0 && amount <= runtimeBalance();
    return cycleExposure + amount <= cycleBudget;
}

function closeCycle() {
    cycleActive = false;
    cycleBudget = 0;
    cycleExposure = 0;
}

        return;

function averageSlope(shortWindow, longWindow) {
    var a = metrics(shortWindow).cappedAvg;
    var b = metrics(longWindow).cappedAvg;
    return a - b;
}

function extremeAlternation(windowSize) {
    var n = Math.min(windowSize, history.length);
    if (n < 3) return 0;

    var arr = history.slice(history.length - n);
    var transitions = 0;
    var eligible = 0;

    for (var i = 1; i < arr.length; i++) {
        var aLow = arr[i - 1] < 1.40;
        var bLow = arr[i] < 1.40;
        var aHigh = arr[i - 1] >= 3.00;
        var bHigh = arr[i] >= 3.00;

        if (aLow || aHigh || bLow || bHigh) {
            eligible++;
            if ((aLow && bHigh) || (aHigh && bLow)) transitions++;
        }
    }

    return eligible ? transitions / eligible : 0;
}

/* ==========================================================
   GLOBAL RECOVERY PLAN
   Trigger: dynamic around 3.0%-4.5% drawdown from the starting credit.
   Goal: restore startBalance, then return to frozen NORMAL mode.
   The new GLOBAL 35/100 cap remains active in Recovery.
   NORMAL hard-stop logic is suspended while Recovery is ACTIVE.
   Recovery ends only after restoring the reference credit. If the platform
   balance cannot fund a wager, the bot keeps observing instead of stopping.
   No RNG: variability comes from the existing Strider sensors.
   ========================================================== */
function normalSessionRounds() {
    return Math.max(0, totalRounds - normalSessionRoundOrigin);
}

/* Dynamic Recovery trigger. It is based on percentage drawdown, not on losing
   1-2-3 bits. Weak/deteriorating structure can bring entry closer to 3.0%;
   strong/stable structure can delay it toward 4.5%. Global Recovery then
   remains active until the original reference balance is fully restored. */
function dynamicRecoveryTriggerPct() {
    if (history.length < WARMUP) return RECOVERY_PLAN_TRIGGER_PCT;

    var m5 = metrics(5);
    var m14 = metrics(14);
    var slope = averageSlope(5, 14);
    var trigger = RECOVERY_PLAN_TRIGGER_PCT;

    if (m14.cappedAvg < 1.85) trigger -= 0.35;
    if (m5.cappedAvg < 1.60) trigger -= 0.20;
    if (slope < -0.35) trigger -= 0.20;
    if (consecutiveLosses >= 2) trigger -= 0.15;

    if (currentPhase === "HOT" && m14.median >= 1.90) trigger += 0.30;
    else if (currentPhase === "POSITIVE" && m14.median >= 1.75 && slope > 0.20) trigger += 0.20;

    return clamp(trigger, 3.00, 4.50);
}

    if (!recoveryPlanActive && dd >= dynamicTrigger) {

        recoveryPlanTargetBalance = startBalance;
        recoveryPlanStartBalance = balance;
        recoveryPlanStartRound = totalRounds;
        recoveryPlanActivations++;
        protectionLevel = 0;
        givebackBrake = 0;
        closeCycle();
        recoveryStep = 0;
        recoveryOriginPhase = null;
        recoveryWaits = 0;
        cooldown = 0;
        log(
            "[RECOVERY PLAN ON]",
            "DD", dd.toFixed(2) + "%",
            "| Dynamic trigger", dynamicTrigger.toFixed(2) + "%",
            "| Balance", bits(balance),
            "| Target", bits(recoveryPlanTargetBalance),
            "| Frequency 35/100: ACTIVE GLOBAL",
            "| Session cap: NONE",
            "| NORMAL 2%/5x limits: SUSPENDED"
        );
        return;
    }

    if (recoveryPlanActive && balance >= recoveryPlanTargetBalance) {
        recoveryPlanActive = false;
        recoveryPlanRecovered++;
        normalSessionRoundOrigin = totalRounds;
        /* Keep rolling-100 wager cap through mode transitions. */
        protectionLevel = 0;
        givebackBrake = 0;

        bestProfitPct = Math.max(0, profitPct(balance));
        closeCycle();
        recoveryStep = 0;
        recoveryOriginPhase = null;
        recoveryWaits = 0;
        cooldown = 0;
        attackLock = 0;
        roundsSinceBet = 99;
        log(
            "[RECOVERY PLAN OFF]",
            "credito ripristinato",
            "| Balance", bits(balance),
            "| Recovery rounds", totalRounds - recoveryPlanStartRound,
            "| NORMAL mode restored"
        );
    }
}

function updateFiveXIntelligence(bust) {
    /* Complete POST3 windows for previous >=5x events. */
    for (var i = 0; i < fiveXEvents.length; i++) {
        var ev = fiveXEvents[i];
        if (ev.round < totalRounds && ev.post.length < FIVE_X_CONTEXT_WINDOW) {
            ev.post.push(bust);
            if (recoveryPlanActive && ev.post.length === FIVE_X_CONTEXT_WINDOW) {
                log("[5X POST]", "event r" + ev.round, "x" + ev.value.toFixed(2), "| post", formatSequence(ev.post));
            }
        }
    }

    if (bust < FIVE_X_THRESHOLD) return;

    var preStart = Math.max(0, history.length - 1 - FIVE_X_CONTEXT_WINDOW);
    var pre = history.slice(preStart, history.length - 1);
    var previous = fiveXEvents.length ? fiveXEvents[fiveXEvents.length - 1] : null;
    var interval = previous ? totalRounds - previous.round : 0;

    fiveXEvents.push({
        round: totalRounds,
        value: bust,
        interval: interval,
        pre: pre,
        post: []
    });

    if (fiveXEvents.length > FIVE_X_MEMORY_MAX) fiveXEvents.shift();

    if (recoveryPlanActive) {
        log("[5X EVENT]", "x" + bust.toFixed(2), "| interval", interval || "first", "| pre", formatSequence(pre));
    }
}

function computeFiveXContext() {
    var events = fiveXEvents.slice();
    var intervals = [];
    var i;

    for (i = Math.max(0, events.length - 25); i < events.length; i++) {
        if (events[i].interval > 0) intervals.push(events[i].interval);
    }

    var lastEvent = events.length ? events[events.length - 1] : null;
    var gap = lastEvent ? Math.max(0, totalRounds - lastEvent.round) : 0;
    var avgInterval = averageNumbers(intervals);
    var medianInterval = medianNumbers(intervals);
    var timing = computeFiveXTimingScore(gap, intervals, medianInterval);

    var currentSeq = history.slice(Math.max(0, history.length - FIVE_X_CONTEXT_WINDOW));
    var preScores = [];
    var postScores = [];

    for (i = Math.max(0, events.length - 25); i < events.length; i++) {
        if (events[i].pre && events[i].pre.length >= 2) {
            preScores.push(sequenceSimilarity5X(currentSeq, events[i].pre));
        }
        if (events[i].post && events[i].post.length >= 2) {
            postScores.push(sequenceSimilarity5X(currentSeq, events[i].post));
        }
    }

    var structure = topAverage(preScores, 5, 50);
    var postStructure = topAverage(postScores, 5, 50);
    var sampleConfidence = clamp(events.length * 5, 0, 100);
    var postZone = !!lastEvent && gap <= FIVE_X_CONTEXT_WINDOW;

    var coherence = timing * 0.45 + structure * 0.45 + sampleConfidence * 0.10;
    if (postZone) coherence -= 14;
    if (postStructure > structure + 10) coherence -= 5;
    coherence = clamp(coherence, 0, 100);

    var expectedRemaining = medianInterval > 0 ? Math.max(0, medianInterval - gap) : 0;

    lastFiveXContext = {
        events: events.length,
        intervals: intervals.length,
        gap: gap,
        avgInterval: avgInterval,
        medianInterval: medianInterval,
        expectedRemaining: expectedRemaining,
        timing: timing,
        structure: structure,
        postStructure: postStructure,
        coherence: coherence,
        postZone: postZone
    };

    return lastFiveXContext;
}

function computeFiveXTimingScore(gap, intervals, medianInterval) {
    if (!intervals.length) return 50;
    if (intervals.length < 3) return 52;

    var scale = Math.max(2, medianInterval * 0.35);
    var weighted = 0;
    var weights = 0;

    for (var i = intervals.length - 1, age = 0; i >= 0; i--, age++) {
        var w = Math.max(0.35, 1 - age * 0.05);
        var d = Math.abs(gap - intervals[i]);
        var similarity = Math.exp(-d / scale);
        weighted += similarity * w;
        weights += w;
    }

    var density = weights ? weighted / weights : 0;
    return clamp(28 + density * 68, 20, 96);
}

function sequenceSimilarity5X(a, b) {
    var n = Math.min(a.length, b.length, FIVE_X_CONTEXT_WINDOW);
    if (n < 2) return 50;

    var score = 0;
    var weights = 0;
    for (var i = 0; i < n; i++) {
        var av = a[a.length - n + i];
        var bv = b[b.length - n + i];
        var bandA = fiveXBand(av);
        var bandB = fiveXBand(bv);
        var bandScore = bandA === bandB ? 100 : (Math.abs(bandA - bandB) === 1 ? 68 : 28);
        var logDistance = Math.abs(Math.log(Math.max(1.01, av)) - Math.log(Math.max(1.01, bv)));
        var numericScore = clamp(100 - logDistance * 48, 0, 100);
        var positionWeight = 0.75 + i * 0.20;
        score += (bandScore * 0.55 + numericScore * 0.45) * positionWeight;
        weights += positionWeight;
    }
    return weights ? score / weights : 50;
}

function fiveXBand(v) {
    if (v < 1.40) return 0;
    if (v < 2.00) return 1;
    if (v < 3.00) return 2;
    if (v < 5.00) return 3;
    return 4;
}

function topAverage(values, topN, fallback) {
    if (!values.length) return fallback;
    var arr = values.slice().sort(function (a, b) { return b - a; });
    var n = Math.min(topN, arr.length);
    var sum = 0;
    for (var i = 0; i < n; i++) sum += arr[i];
    return sum / n;
}

function averageNumbers(values) {
    if (!values.length) return 0;
    var sum = 0;
    for (var i = 0; i < values.length; i++) sum += values[i];
    return sum / values.length;
}

function medianNumbers(values) {
    if (!values.length) return 0;
    var arr = values.slice().sort(function (a, b) { return a - b; });
    var n = arr.length;
    if (n % 2) return arr[(n - 1) / 2];
    return (arr[n / 2 - 1] + arr[n / 2]) / 2;
}

function formatSequence(values) {
    if (!values || !values.length) return "[]";
    var out = [];
    for (var i = 0; i < values.length; i++) out.push(Number(values[i]).toFixed(2));
    return "[" + out.join(",") + "]";
}

function computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty) {
    var score = confidence * 0.42 + targetPotential * 0.28 + 18;

    if (currentPhase === "POSITIVE") {
        if (s.m14.median >= 1.65) score += 6;
        if (s.m14.p200 >= 0.35) score += 5;
        if (s.m5.p140 >= 0.60) score += 4;
        if (s.m14.median < 1.40) score -= 8;
    } else if (currentPhase === "HOT") {
        if (s.m14.median >= 2.00) score += 7;
        if (s.m14.p300 >= 0.22) score += 6;
        if (s.m5.p200 >= 0.50) score += 5;
        if (rawPhase !== "HOT") score -= 8;
    } else {
        if (breakWatchRounds > 0) score += clamp(breakWatchStrength * 0.10, 0, 10);
        else score -= 10;
        if (s.lowStreak >= 3) score += 3;
    }

    if (s.recentSlope > 0.25) score += 5;
    if (s.recentSlope < -0.25) score -= 7;
    if (hotTransitionCaution) score -= 8;

    score -= volatilityPenalty * 0.35;
    /* Recent losses matter, but Recovery must not become permanently paralysed
       by the NORMAL feedback penalty. */
    score -= lastFeedback.basePenalty * 0.15;
    score -= clamp(lastFeedback.lossStreak * 1.5, 0, 6);

    /* 5X intelligence is an additional recognition lens, never a prediction. */
    var fx = computeFiveXContext();
    if (fx.events >= 3) {
        if (fx.coherence >= 88 && fx.timing >= 75 && fx.structure >= 78) score += 12;
        else if (fx.coherence >= 80) score += 8;
        else if (fx.coherence >= 70) score += 4;
        else if (fx.coherence < 40) score -= 3;
        if (fx.postZone) score -= 4;
    }

    return clamp(score, 0, 100);
}

function chooseRecoveryPlanPayout(s, confidence, targetPotential, pace, attack, recovery) {
    var volatilityPenalty = computeVolatilityPenalty(s);
    var quality = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    var fx = lastFiveXContext;
    var p;

    if (currentPhase === "NEGATIVE") {
        p = 1.08 + quality / 100 * 0.62;
        if (breakWatchRounds > 0) p += breakWatchStrength / 100 * 0.35;
    } else if (currentPhase === "POSITIVE") {
        p = 1.15 + quality / 100 * 1.10 + s.m14.p200 * 0.45;
    } else {
        p = 1.30 + quality / 100 * 1.55 + s.m14.p300 * 0.85;
    }

    /* When interval + structure strongly agree, Recovery may deliberately
       choose targets above the NORMAL 5x ceiling. */
    if (fx.events >= 3) {
        if (fx.coherence >= 70) p += (fx.coherence - 65) / 100 * 2.5;
        if (fx.coherence >= 84 && fx.timing >= 72 && fx.structure >= 78) p += 1.10;
        if (fx.coherence >= 92 && quality >= 90) p += 1.80;
        if (fx.postZone) p -= 0.30;
    }

    if (pace === "SLOW") p -= 0.12;
    if (pace === "AGGRESSIVE") p += 0.25;
    if (attack) p += 0.45;
    if (recovery) p -= 0.12;
    if (hotTransitionCaution) p -= 0.15;

    /* No NORMAL 5x clamp here. The formula itself determines the target. */
    return roundMultiplier(Math.max(MIN_PAYOUT, p));
}

function chooseRecoveryPlanBetSize(s, confidence, targetPotential, pace, attack, recovery) {
    var balance = runtimeBalance();
    if (balance <= unit) return unit;

    var volatilityPenalty = computeVolatilityPenalty(s);
    var quality = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    var fx = lastFiveXContext;
    var fraction = 0.010;

    if (quality >= 72) fraction = 0.015;
    if (quality >= 78) fraction = 0.025;
    if (quality >= 84) fraction = 0.040;
    if (quality >= 89) fraction = 0.060;
    if (quality >= 94) fraction = 0.080;

    if (fx.events >= 3 && fx.coherence >= 78) fraction += 0.010;
    if (fx.events >= 3 && fx.coherence >= 88 && fx.structure >= 80) fraction += 0.015;

    if (attack) fraction *= 1.18;
    if (recovery) fraction *= recoveryStep === 2 ? 1.08 : 0.96;
    if (pace === "SLOW") fraction *= 0.70;
    if (pace === "AGGRESSIVE") fraction *= 1.12;
    if (volatilityPenalty >= 24) fraction *= 0.78;
    if (lastFeedback.basePenalty >= 18) fraction *= 0.82;

    var gapRatio = recoveryPlanTargetBalance > 0
        ? Math.max(0, recoveryPlanTargetBalance - balance) / recoveryPlanTargetBalance
        : 0;
    if (gapRatio >= 0.25 && quality >= 86) fraction *= 1.08;
    if (gapRatio >= 0.50 && quality >= 90) fraction *= 1.10;

    /* Capital-adaptive Recovery. Smaller balances keep the same intelligence
       and persistence, but the individual wager is scaled down. This avoids
       treating 729 bits like a 4000-bit bankroll. */
    fraction *= capitalRiskScale(balance);
    fraction = clamp(fraction, 0.004, recoveryMaxFractionForCapital(balance));

    /* Free adaptive amount in whole bits, not tied to NORMAL 2% cycle budget.
       It remains quality-scaled rather than a fixed martingale. */
    var amount = Math.floor(balance * fraction);
    amount = roundSats(amount);
    amount = Math.max(unit, amount);
    amount = Math.min(amount, roundSats(Math.max(unit, balance - unit)));
    return amount;
}

function chooseRecoveryPlanPace(s, confidence, targetPotential, volatilityPenalty) {
    var q = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    lastRecoveryPlanSignal = q;
    if (q >= 84) return "AGGRESSIVE";
    if (q >= 72) return "NORMAL";
    return "SLOW";
}

function evaluateRecoveryPlanEntry(s, confidence, targetPotential, volatilityPenalty, pace, attackBlocked) {
    var quality = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    lastRecoveryPlanSignal = quality;
    var fx = lastFiveXContext;

    var threshold = currentPhase === "NEGATIVE" ? 74 : (currentPhase === "HOT" ? 68 : 66);
    if (volatilityPenalty >= 24) threshold += 3;
    if (lastFeedback.basePenalty >= 18) threshold += 2;
    if (lastFeedback.basePenalty >= 22) threshold += 1;
    if (hotTransitionCaution) threshold += 3;
    if (fx.events >= 3 && fx.coherence >= 82 && fx.timing >= 70 && fx.structure >= 76) threshold -= 2;
    if (fx.events >= 3 && fx.postZone) threshold += 2;

    if (quality < threshold) {
        return {
            enter: false,
            reason: "Recovery Plan quality " + quality.toFixed(0) + " < " + threshold.toFixed(0)
        };
    }

    var attack = false;
    var attackConfidence = confidence - lastFeedback.attackPenalty;
    if (!attackBlocked && quality >= 82 && attackConfidence >= 76 && targetPotential >= 72) {
        if (currentPhase === "HOT" || (currentPhase === "POSITIVE" && s.m14.p200 >= 0.40)) {
            attack = true;
        }
    }
    if (!attackBlocked && fx.events >= 3 && fx.coherence >= 88 && fx.structure >= 82 && quality >= 86) {
        attack = true;
    }

    var payout = chooseAdaptivePayout(s, confidence, targetPotential, pace, attack, false);
    var amount = chooseAdaptiveBetSize(s, confidence, targetPotential, pace, attack, false);

    return {
        enter: true,
        mode: attack ? "ATTACK" : "BASE",
        payout: payout,
        amount: amount,
        reason: "Recovery Plan validated Q" + quality.toFixed(0)
    };
}

/* ==========================================================
   CAPITAL ADAPTIVE HELPERS
   ========================================================== */
function balanceInBits(balance) {
    balance = Number(balance);
    if (!isFinite(balance) || balance <= 0) return 0;
    return balance / 100;
}

function capitalRiskScale(balance) {
    var b = balanceInBits(balance);
    if (b <= 0) return 0.55;
    /* At 729 bits => 0.729. At >=1000 bits => original 2.0 sizing. */
    return clamp(b / CAPITAL_REFERENCE_BITS, 0.55, 1.00);
}

function recoveryMaxFractionForCapital(balance) {
    var b = balanceInBits(balance);
    if (b < 500) return 0.022;
    if (b < 1000) return 0.028;
    if (b < 2000) return 0.040;
    if (b < 5000) return 0.060;
    return 0.080;
}

   BALANCE / TARGETS
   ========================================================== */
function updateBestBalance(balance) {
    if (balance > bestBalance) bestBalance = balance;
    var p = profitPct(balance);
    if (p > bestProfitPct) bestProfitPct = p;
}

function profitPct(balance) {
    if (startBalance <= 0) return 0;
    return ((balance - startBalance) / startBalance) * 100;
}

function drawdownPctFromStart(balance) {
    if (startBalance <= 0) return 0;
    return Math.max(0, ((startBalance - balance) / startBalance) * 100);
}

function ksEmpty(){return{n:0,w:0,stake:0,pnl:0,maxBust:0,maxSelected:0};}
function ksiBucket(x){return x<1.25?"Z":x<2?"N":x<3?"P":x<10?"H":"X";}
function ksiTailKey(n){return KSI_DIGITS.slice(Math.max(0,KSI_DIGITS.length-n)).join("");}

function ksiMedian(a){if(!a.length)return 0;var q=a.slice().sort(function(x,y){return x-y;}),k=Math.floor(q.length/2);return q.length%2?q[k]:(q[k-1]+q[k])/2;}
function ksiZero(t){
    var h=KSI_HIST.slice(-400),cnt=0,last=-1;
    for(var i=0;i<h.length;i++)if(h[i]>=t){cnt++;last=i;}
    var g=last<0?h.length:h.length-1-last;
    var empirical=(cnt+ksiBase(t)*40)/(h.length+40);
    var interval=cnt? h.length/cnt:0;
    return{count:cnt,gap:g,medianGap:interval,frequency:empirical};
}
function ksiPredict(t){
    var base=ksiBase(t),num=base*80,den=80,support=0,detail=[];
    for(var n=3;n<=5;n++){
        if(KSI_DIGITS.length<n)continue;
        var key=ksiPatternKey(n),slot=KSI_PATTERN[key];
        if(!slot)continue;
        var hit=slot.hits[String(t)]||0;
        var local=(hit+base*35)/(slot.n+35),weight=Math.min(50,slot.n)*(n-2);
        num+=local*weight;den+=weight;support+=slot.n;
        detail.push(key+"="+slot.n);
    }
    return{base:base,p:num/den,support:support,detail:detail.join(",")};
}
function ksiHf32(){
    var h=KSI_HIST.slice(-32),n=h.length,p2=0,p3=0,avg=0;
    for(var i=0;i<n;i++){p2+=h[i]>=2?1:0;p3+=h[i]>=3?1:0;avg+=Math.min(h[i],10);}
    return {score:n?clamp(50+((p2/n)-.5)*40+((p3/n)-.33)*22+((avg/n)-2.2)*7,0,100):50,
            p2:n?p2/n:0,p3:n?p3/n:0};
}
function ksiChooseHunt(snapshot){
    var options=[],hf=ksiHf32();
    if(KSI_HIST.length<Math.max(100,WARMUP))return null;
    for(var i=0;i<KSI_HUNT_TARGETS.length;i++){
        var t=KSI_HUNT_TARGETS[i],pred=ksiPredict(t),gap=ksiZero(t);
        var base=pred.base,delta=(pred.p-base)*t;
        var supportScore=Math.min(14,pred.support*.12);
        var score=clamp(50+supportScore+clamp(delta*25,-12,12)+(hf.score-50)*.12,0,100);
        var stats=KSI_AT_TARGET[String(t)]||ksEmpty();
        // These are evidence/ranking bands, not probabilities of next outcome.
        var minSup=t>=50?70:(t>=20?40:(t>=12?24:8));
        if(pred.support<minSup || score<54)continue;
        // Avoid continuously showing the same long shot just for elapsed gap.
        if(t>=20 && gap.count<2)continue;
        options.push({target:t,score:score,pred:pred,zero:gap,stats:stats,hf:hf});
    }
    options.sort(function(a,b){return b.score-a.score || a.target-b.target;});
    return options.length?options[0]:null;
}
function ksiAttackRoll(){

    return{n:a.length-from,stake:stake,pnl:pnl,wr:a.length>from?won/(a.length-from):0,
           roi:stake?pnl/stake:0};
}

function ksiPrepareRound(snapshot){
    KSI_CHECKS++;
    KSI_EXPLAIN.lastAttack="";
    KSI_HUNT_PENDING=null;
    if(config.enableHunt.value===true){
        while(KSI_HUNT_MARKS.length && KSI_HUNT_MARKS[0] <= totalRounds-100)KSI_HUNT_MARKS.shift();
        var huntSpaced = !KSI_HUNT_MARKS.length || totalRounds-KSI_HUNT_MARKS[KSI_HUNT_MARKS.length-1] >= KSI_HUNT_SPACING;
        var huntCapped = KSI_HUNT_MARKS.length >= KSI_HUNT_MAX_PER_100;
        var h=(huntSpaced && !huntCapped)?ksiChooseHunt(snapshot):null;
        if(h){
            KSI_HUNT_MARKS.push(totalRounds);
            KSI_HUNT_PENDING={target:h.target,score:h.score,
                phase:currentPhase,pat:h.pred.detail,support:h.pred.support,
                hf:h.hf.score,gap:h.zero.gap};
            KSI_EXPLAIN.lastHunt=h.target+"x SCORE "+h.score.toFixed(1)+
                " SUPPORT "+h.pred.support+" GAP "+h.zero.gap+" HF32 "+h.hf.score.toFixed(1);
        } else KSI_EXPLAIN.lastHunt=huntCapped?"HUNT limit 15/100":(huntSpaced?"Nessun target con evidenza minima":"HUNT spacing");
    }
}

function ksiSettleLearning(bust){
    if(!isFinite(bust)||bust<1)return;
    // Brier score is evaluated BEFORE including this bust in the training data.
    if(KSI_DIGITS.length>=5){
        var pred=ksiPredict(3),outcome=bust>=3?1:0;
        KSI_ML.checks++;
        KSI_ML.brier+=(pred.p-outcome)*(pred.p-outcome);
        KSI_ML.baseline+=(pred.base-outcome)*(pred.base-outcome);
    }
    if(KSI_HUNT_PENDING){
        var hp=KSI_HUNT_PENDING,win=bust>=hp.target,pnl=win?Math.round(100*(hp.target-1)):-100;
        var z=KSI_HUNT_SHADOW;z.checks++;z.stake+=100;z.pnl+=pnl;
        if(win)z.wins++;
        z.selectedMax=Math.max(z.selectedMax,hp.target);
        z.bustMax=Math.max(z.bustMax,bust);
        var at=KSI_AT_TARGET[String(hp.target)]||(KSI_AT_TARGET[String(hp.target)]=ksEmpty());
        at.n++;if(win)at.w++;at.pnl+=pnl;at.stake+=100;
        if(totalRounds%100===0){
            log("[HUNT SHADOW]",hp.target+"x","|",win?"WIN":"LOSS",
                "| P/L",signedBits(pnl),"| PAT",hp.pat,"| support",hp.support,
                "| gap",hp.gap,"| HF32",hp.hf.toFixed(0));
        }
        KSI_HUNT_PENDING=null;
    }
    for(var n=3;n<=5;n++){
        if(KSI_DIGITS.length<n)continue;
        var key=ksiPatternKey(n),m=KSI_PATTERN[key];
        if(!m)KSI_PATTERN[key]=m={n:0,hits:Object.create(null)};
        m.n++;
        for(var j=0;j<KSI_TARGETS.length;j++){
            var t=KSI_TARGETS[j],k=String(t);
            if(bust>=t)m.hits[k]=(m.hits[k]||0)+1;
        }
        KSI_EXPLAIN.lastPattern=key;
    }
    KSI_HIST.push(bust);KSI_DIGITS.push(ksiBucket(bust));
    if(KSI_HIST.length>KSI_MAX_HISTORY){KSI_HIST.shift();KSI_DIGITS.shift();}
    if(bust>=3)KSI_HIGH_EVENTS["3"]=(KSI_HIGH_EVENTS["3"]||0)+1;
    if(bust>=5)KSI_HIGH_EVENTS["5"]=(KSI_HIGH_EVENTS["5"]||0)+1;
    if(bust>=10)KSI_HIGH_EVENTS["10"]=(KSI_HIGH_EVENTS["10"]||0)+1;
    if(bust>=20)KSI_HIGH_EVENTS["20"]=(KSI_HIGH_EVENTS["20"]||0)+1;
    if(bust>=50)KSI_HIGH_EVENTS["50"]=(KSI_HIGH_EVENTS["50"]||0)+1;
    if(bust>=100)KSI_HIGH_EVENTS["100"]=(KSI_HIGH_EVENTS["100"]||0)+1;
}

function k22BeginObserve(why){

    K22_STARTED_OBSERVE=true;
    K22_OBSERVE_UNTIL=Math.max(K22_OBSERVE_UNTIL,totalRounds+Math.max(40,Math.min(50,Number(config.observationRounds.value)||45)));
    recoveryStep=0;recoveryWaits=0;closeCycle();
    k22StateLog(why+' | sola osservazione fino al round '+K22_OBSERVE_UNTIL);
}
function k22QualityForProbe(s){
    var hf=ksiHf32(),r=ksiAttackRoll(),fx=computeFiveXContext();
    // Filtro diagnostico: un segnale non equivale a vantaggio accertato.
    var modelReady=KSI_ML.checks>=150 && KSI_ML.brier<KSI_ML.baseline;
    var baselineOk=r.n>=20 && r.roi>0;
    var ordinary=s.m14.median>=1.65 && s.m30.median>=1.45 && s.m14.p200>=0.40 && hf.score>=52;
    var sentinelRecovery=fx.events>=3&&fx.coherence>=78&&fx.structure>=70&&
       s.m14.median>=1.50&&s.m30.median>=1.40&&hf.score>=48;
    return currentPhase!=='NEGATIVE' && modelReady && baselineOk &&
           (ordinary || sentinelRecovery);
}
function k22UpdateState(balance,s){
    var dd=k22PctDrawdown();
    K22_RECOVERY_TARGET_BITS=Math.max(0,(startBalance-balance)/300); // 1/3 della perdita, in bit
    if(K22_HARD_LOCK || dd>=Math.max(2.5,Number(config.maxRiskPct.value)||2.5)){
        K22_HARD_LOCK=true; K22_STATE='HARD_STOP'; recoveryStep=0; closeCycle();
        k22StateLog('HARD_STOP '+dd.toFixed(2)+'%: osservazione indefinita, riavvio manuale');
        return;
    }
    var observeAt=Math.min(1.98,Number(config.observeDdPct.value)||1.98);
    if(dd>=observeAt){
        if(!K22_STARTED_OBSERVE){k22BeginObserve('Drawdown '+dd.toFixed(2)+'%');}
        if(K22_STATE==='PROBE' && K22_PROBE_SHOTS>=2){
            K22_PROBE_BLOCKS++;K22_PROBE_SHOTS=0;k22BeginObserve('Completata tranche prudente '+K22_PROBE_BLOCKS);
        }
        if(K22_PROBE_BLOCKS>=Math.min(5,Number(config.maxRecoveryBlocks.value)||5)){
            K22_STATE='HARD_STOP';K22_HARD_LOCK=true;closeCycle();
            k22StateLog('5 tranche esaurite: solo osservazione');return;
        }
        if(K22_STATE==='OBSERVE' && totalRounds>=K22_OBSERVE_UNTIL && k22QualityForProbe(s)){
            K22_STATE='PROBE';K22_SHOT_GOAL=K22_PROBE_BLOCKS+1;
            closeCycle(); k22StateLog('PROBE abilitata: max due sondaggi da 1 bit');
        }
        return;
    }
    if(K22_STARTED_OBSERVE){K22_STARTED_OBSERVE=false;K22_PROBE_SHOTS=0;K22_OBSERVE_UNTIL=0;}
    if(totalRounds<K22_PROFIT_LOCK_UNTIL){K22_STATE='PROFIT_LOCK';return;}
    K22_STATE=dd>=Math.min(1.2,Number(config.cautionDdPct.value)||1.2)?'CAUTION':'NORMAL';
}
function k22PreBetGate(balance,s){
    if(K22_STATE==='OBSERVE'||K22_STATE==='HARD_STOP'||K22_STATE==='PROFIT_LOCK'){
        if(totalRounds%25===0)log('[OBSERVE]',K22_STATE,'| round',totalRounds,'| DD',k22PctDrawdown().toFixed(2),'| monitoro pattern e target');
        return false;
    }
    if(K22_STATE==='PROBE' && (K22_PROBE_SHOTS>=2 || totalRounds-K22_LAST_PROBE_ROUND<k22RecoveryGap()))return false;
    return true;
}
function k22BestRollingMode(){
    var modes=['BASE','ATTACK'],best='',bestScore=-1e9;
    for(var i=0;i<modes.length;i++){
        var m=modes[i],a=KSI_ROLL[m],n=Math.min(a.length,40),stake=0,pnl=0;
        for(var j=a.length-n;j<a.length;j++){stake+=a[j].stake;pnl+=a[j].pnl;}
        if(n<25||!stake)continue;
        var roi=pnl/stake,score=roi*Math.sqrt(n);
        if(score>bestScore){best=m;bestScore=score;}
    }
    return bestScore>0?best:'';
}
/* V3.0: dedicated conservative PROBE path. This intentionally DOES NOT
   require the BASE/ATTACK entry or targetPotential >65. Risk/probe/ML remain. */
function k30RecoveryProbe(s,conf,tp,pace,volatilityPenalty){
    if(K22_STATE!=='PROBE')return false;
    var best=K24_LAST_PLAN && K24_LAST_PLAN.best;
    var why='';
    if(cooldown>0){cooldown--;why='COOLDOWN';}
    else if(decisionWindow.length>=20 && recentBetCount()>=MAX_BETS_PER_100)why='FREQUENCY_35_100';
    else if(!k22QualityForProbe(s))why='QUALITY_NOT_VALID';
    else if(!best || best.family!=='RECOVERY' || !best.approved)why='RECOVERY_ML_NOT_APPROVED';
    else if(conf<72)why='CONFIDENCE_BELOW_72';
    else if(K22_PROBE_SHOTS>=2)why='TRANCHE_COMPLETED';
    else if(totalRounds-K22_LAST_PROBE_ROUND<k22RecoveryGap())why='PROBE_SPACING';
    if(why){log('[V30 PROBE WAIT]',why,'| TP',tp.toFixed(2),'| ML',best?best.id:'NONE');return true;}
    var proposal={enter:true,mode:'RECOVERY_'+(K22_PROBE_SHOTS+1),amount:unit,
      payout:Math.min(2,Math.max(1.01,best.target)),reason:'V30_RECOVERY_PROBE_ML '+best.id};
    proposal=K26_GOVERNOR.gate(proposal,K24_LAST_PLAN);
    if(!proposal.enter){log('[V30 PROBE WAIT]',proposal.reason);return true;}
    startCycleIfNeeded();
    if(!k22AllowMoney(unit,proposal.mode)||!canUseCycleBudget(unit)||!canBet(unit)){
      log('[V30 PROBE WAIT] CAPITAL_OR_CYCLE_GATE');return true;
    }
    // The probe is one of the ensemble selections, never an extra parallel bet.
    K27_ENSEMBLE.markDirect(proposal.mode,proposal.payout,'RECOVERY_ML_PROBE');
    K30_DRIFT.plan(proposal);
    lastMode=proposal.mode;
    placeBet(unit,proposal.payout,proposal.mode,s,conf,tp,'SLOW');
    if(!playedThisRound)K27_ENSEMBLE.noteVeto('PROBE_BET_NOT_PLACED');
    return true;
}
function k22ReviewEntry(e,s,conf,tp){
    if(K22_STATE==='PROBE'){
        if(!k22QualityForProbe(s) || !e.enter || conf<72 || tp<65 ||
           (!K24_LAST_PLAN || !K24_LAST_PLAN.best || K24_LAST_PLAN.best.family!=='RECOVERY')){
          return{enter:false,reason:'PROBE aspetta conferme / ML o contesto non favorevole'};
        }
        if(totalRounds-K22_LAST_PROBE_ROUND<k22RecoveryGap())return {enter:false,reason:'PROBE spacing 4-7 round'};
        var pay=chooseAdaptivePayout(s,conf,tp,'SLOW',false,true);
        return{enter:true,mode:'RECOVERY_'+(K22_PROBE_SHOTS+1),amount:unit,
          payout:Math.min(pay,2),reason:'sonda cauta 1 bit ('+(K22_PROBE_SHOTS+1)+'/2)'};
    }
    if(!e.enter)return e;
    var chosen=k22BestRollingMode();
    if(chosen==='BASE' && e.mode==='ATTACK'){
        e.mode='BASE';K22_STRATEGY_TALLY.attackBlocks++;
        e.amount=chooseAdaptiveBetSize(s,conf,tp,'SLOW',false,false);
        e.payout=chooseAdaptivePayout(s,conf,tp,'SLOW',false,false);
        e.reason='ML preferisce BASE: ATTACK ROI inferiore';
        K22_STRATEGY_TALLY.mlVotes++;
    }else if(chosen==='ATTACK' && e.mode==='BASE' && currentPhase==='HOT' && conf>=75 && tp>=75 && !ksiGuardAttack()){
        e.mode='ATTACK';e.amount=chooseAdaptiveBetSize(s,conf,tp,'NORMAL',true,false);
        e.payout=chooseAdaptivePayout(s,conf,tp,'NORMAL',true,false);
        e.reason='ML ATTACK validato dal ROI rolling';
        K22_STRATEGY_TALLY.mlVotes++;
    }
    if(K22_STATE==='CAUTION'){
        e.amount=unit;
        if(e.mode==='ATTACK' && chosen!=='ATTACK')e.mode='BASE';
    }
    return e;
}
function k22AllowMoney(amount,mode){
    amount=roundSats(amount);
    if(K22_STATE==='OBSERVE'||K22_STATE==='HARD_STOP'||K22_STATE==='PROFIT_LOCK')return false;
    if(K22_STATE==='PROBE' && (mode.indexOf('RECOVERY')!==0||amount!==unit||K22_PROBE_SHOTS>=2))return false;
    if(mode.indexOf('RECOVERY')===0 && amount!==unit)return false;
    if(amount>8*100)return false;
    if(k22PctDrawdown()>=2.5)return false;
    // Un prezzo dello stake elevato NON è reso lecito da una perdita precedente.
    return true;
}
function k22ReviewLength(){
    var r=ksiAttackRoll();
    if(K22_STATE==='CAUTION' || K22_STATE==='OBSERVE')return 90;
    if(currentPhase==='HOT' && r.n>=20 && r.roi>0.03)return 110;
    return 100;
}
function k22SessionReview(){
    var window=k22ReviewLength();
    if(totalRounds-K22_LAST_REVIEW_ROUND<window)return;
    var r=ksiAttackRoll();
    K22_LAST_REVIEW_ROUND=totalRounds;K22_REVIEW_COUNT++;
    log('[90-110 REVIEW]',window,'round | STATO',K22_STATE,
        '| P/L',signedBits(runtimeBalance()-startBalance),
        '| ATTACK ROI',r.n?(100*r.roi).toFixed(2)+'%':'N/A',
        '| ML Brier',KSI_ML.checks?(KSI_ML.brier/KSI_ML.checks).toFixed(5):'N/A');
}
function k22RoundFeedback(bet,pnl){
    var a=ksiAttackRoll();
    var poor=a.n>=12 && a.roi<-0.12;
    if(poor)K22_POOR_ROUNDS++;else K22_POOR_ROUNDS=0;
    if(K22_POOR_ROUNDS>=45 && K22_STATE==='NORMAL'){
        K22_STRATEGY_TALLY.poorStops++;
        K22_PROFIT_LOCK_UNTIL=Math.max(K22_PROFIT_LOCK_UNTIL,totalRounds+45);
        K22_POOR_ROUNDS=0;
        k22StateLog('ROI ATTACK negativo per 45 round, osservazione prudente');
    }
    if(bet && K22_STATE==='PROBE'){
        if(pnl>0)K22_PROBE_WIN_BITS+=pnl/100;
        else K22_PROBE_LOSS_BITS+=pnl/100;
    }
}
function k22AfterRound(){
    var dd=k22PctDrawdown(),p=profitPct(runtimeBalance());
    if(p>=5 && K22_LAST_TARGET_LOCK<5){
        K22_LAST_TARGET_LOCK=5;K22_STRATEGY_TALLY.profitLocks++;
        K22_PROFIT_LOCK_UNTIL=Math.max(K22_PROFIT_LOCK_UNTIL,totalRounds+45);
        closeCycle(); k22StateLog('HOT goal +5% raggiunto: pausa protezione profitti');
    }else if(p>=2 && K22_LAST_TARGET_LOCK<2){
        K22_LAST_TARGET_LOCK=2;K22_STRATEGY_TALLY.profitLocks++;
        K22_PROFIT_LOCK_UNTIL=Math.max(K22_PROFIT_LOCK_UNTIL,totalRounds+20);
        closeCycle(); k22StateLog('Goal +2% raggiunto: pausa di protezione');
    }
    if(dd>=1.98 && K22_STATE==='PROBE' && K22_PROBE_SHOTS>=2){
      K22_PROBE_BLOCKS++;K22_PROBE_SHOTS=0;k22BeginObserve('Due sonde chiuse; riparte osservazione');
    }
}

/* ==========================================================
   UTILITIES
   ========================================================== */
function roundSats(value) {
    // LIVE FIX: always return whole bits (100 satoshis per bit).
    // Floor is deliberate so risk never exceeds the calculated amount.
    value = Number(value);
    if (!isFinite(value)) return 100;
    return Math.max(100, Math.floor(value / 100) * 100);
}

function roundMultiplier(value) {
    return Math.round(value * 100) / 100;
}

function bits(sats) {
    return (sats / 100).toFixed(2);
}

function signedBits(sats) {
    var value = sats / 100;
    return (value >= 0 ? "+" : "") + value.toFixed(2);
}

function pct(value) {
    return (value * 100).toFixed(1) + "%";
}

function clamp(value, min, max) {
    return Math.max(min, Math.min(max, value));
}

   V2.3 CENTRAL STRATEGY GOVERNOR

   una vincita saltata non e' prova che si doveva puntare.

function k24ActualArm(mode,payout){
    if(mode==='PATTERN')return 'PATTERN_'+Math.round(payout*100);
    if(mode==='CLASSIC21')return 'CLASSIC_'+String(K27_ACTIVE_CLASSIC_ID||'GEN').replace('-','_');
    if(mode==='DEEP_STRIKE')return 'DEEP_'+Math.round(payout);
    var id='BASE_135',d=1e9,arms=K24_BRAIN.arms;
    for(var i=0;i<arms.length;i++){
        var a=arms[i];
        var fam=mode.indexOf('RECOVERY')===0?'RECOVERY':mode==='HUNT'?'SENTINEL':mode;
        if(a.family!==fam)continue;
        var dist=Math.abs(a.target-payout);
        if(dist<d){d=dist;id=a.id;}
    }
    return id;
}
function k24Supervise(e,s,conf,tp,pace,attackBlocked){
    if(!config.mlSupervisor.value)return e;
    // PATTERN: il riconoscimento numerico e' il trigger; l'affidabilita' successiva
    // autorizza o nega denaro, come per le altre strategie. Non confondere i due.
    var pp=K25_LAST_PATTERN&&K25_LAST_PATTERN.selected;
    if(pp && K22_STATE!=='OBSERVE' && K22_STATE!=='HARD_STOP' &&
        K22_STATE!=='PROFIT_LOCK' && K22_STATE!=='PROBE' &&
        (!STRIKER_LIVE || config.patternLive.value===true) && givebackBrake!==2){
        var pat=pp.bestValidated, original=K24_LAST_PLAN&&K24_LAST_PLAN.best;
        // Una strategia approvata prevale solo con margine prudente migliore.
        if(!original || pat.conservativeEV>original.score+0.015){
          K24_STATS.promote++;
          return {enter:true,mode:'PATTERN',amount:unit,payout:pat.target,
            reason:'V25 PATTERN '+pp.pattern+' at step +'+pp.horizon+
              ' train='+pat.train+' eval='+pat.eval+' lowerROI='+pat.conservativeEV.toFixed(3)};
        }
    }
    if(!K24_LAST_PLAN || !K24_LAST_PLAN.best){
        if(e.enter)K24_STATS.veto++;
        return {enter:false,reason:'ML V24: manca evidenza indipendente e validata, solo osservazione'};
    }
    var best=K24_LAST_PLAN.best;
    if(K22_STATE==='HARD_STOP'||K22_STATE==='OBSERVE'||K22_STATE==='PROFIT_LOCK'){
        return {enter:false,reason:'V24 RISK '+K22_STATE};
    }
    if(K22_STATE==='PROBE'){
        if(best.family!=='RECOVERY' || !e.enter || conf<72){
            K24_STATS.probesRejected++;
            return{enter:false,reason:'V24 recovery senza validazione'};
        }
        e.amount=unit;
        e.mode='RECOVERY_1';
        return e;
    }
    if(best.family==='RECOVERY')return{enter:false,reason:'V24 recovery fuori contesto'};
    if(best.family==='SENTINEL'){
        if(!STRIKER_LIVE){
            var hs=ksiChooseHunt(s);
            if(!config.enableHunt.value||!hs||Math.abs(hs.target-best.target)>.001||!e.enter||attackBlocked){
                return{enter:false,reason:'V28 SENTINEL SHADOW: nessuna doppia conferma'};
            }
            K24_STATS.swap++;
            return{enter:true,mode:'HUNT',payout:best.target,amount:unit,reason:'V28 SENTINEL SHADOW con doppia conferma'};
        }
        if(!config.mlSentinelLive.value||!config.huntLive.value){
            K24_STATS.huntRejected++;
            return{enter:false,reason:'V24 hunt non autorizzato LIVE'};
        }
        var h=ksiChooseHunt(s);
        if(!h||!ksiHuntMayGoLive(h)||Math.abs(h.target-best.target)>.001){
            return{enter:false,reason:'V24 hunt senza doppia conferma'};
        }
        if(!e.enter||attackBlocked)return{enter:false,reason:'V24 hunt sospeso da controllo STRIDER'};
        K24_STATS.swap++;
        return{enter:true,mode:'HUNT',payout:best.target,amount:unit,reason:'V24 SENTINEL validato + doppia conferma'};
    }
    if(best.family==='ATTACK' && (currentPhase==='NEGATIVE'||attackBlocked||conf<73)){
        return{enter:false,reason:'V24 ATTACK sospeso dal controllo fase/ROI'};
    }
    if(!e.enter){
        // Promozione prudente solo se le regole operative del bot non hanno imposto pausa.
        if(currentPhase==='NEGATIVE'||attackBlocked||conf<72||tp<45||givebackBrake===2){
            return e;
        }
        K24_STATS.promote++;
        return {enter:true,mode:best.family,amount:unit,payout:best.target,
            reason:'V24 conferma causale e sicurezza: '+best.id};
    }
    if(e.mode!==best.family||Math.abs(e.payout-best.target)>.001){
        K24_STATS.swap++;
        e.mode=best.family;e.payout=best.target;
        e.reason='V24 scelta validata: '+best.id;
    }
    // In prudenza e Recovery mai moltiplicare le puntate.
    e.amount=unit;
    return e;
}

    if(st==='OBSERVE'||st==='HARD_STOP'||st==='PROFIT_LOCK')return false;
    if(st==='PROBE')return a.family==='RECOVERY'&&Number(c.confidence||0)>=72;
    if(a.family==='RECOVERY')return false;
    if(st==='CAUTION'&&(a.family==='ATTACK'||a.family==='SENTINEL'))return false;
    if(a.family==='ATTACK'&&(Number(c.confidence||0)<70||c.phase==='NEGATIVE'))return false;
    if(a.family==='SENTINEL'&&!c.huntAllowed)return false;
    return true;
  }
  function decStat(z,w){z.n=z.n*DECAY+1;z.w=z.w*DECAY+(w?1:0);z.pnl=z.pnl*DECAY+(w?z.target-1:-1);}
  function confidenceLimit(p,n,z){
    // One-sided Wilson lower bound, valido per esiti binari e conservativo.
    var den=1+z*z/n,ctr=p+z*z/(2*n),rad=z*Math.sqrt((p*(1-p)+z*z/(4*n))/n);
    return Math.max(0,(ctr-rad)/den);
  }
  function candidates(c){
    if(pending)throw Error('V24: esito del round precedente non registrato');
    if(!c||!Number.isFinite(c.med14)||!Number.isFinite(c.p2))throw Error('V24: contesto invalido');
    var k=key(c),loc=makeLoc(k),locCal=contextCal[k],all=[],best=null,paper=null;
    for(var i=0;i<ARMS.length;i++){
      var a=ARMS[i],g=globals[i],l=loc[i],p0=.99/a.target;
      var pg=(g.w+GLOBAL_PRIOR*p0)/(g.n+GLOBAL_PRIOR);
      var pl=(l.w+LOCAL_PRIOR*pg)/(l.n+LOCAL_PRIOR);
      var cc=locCal[i],cg=calibrations[i];
      // Verifica causale: previsioni storiche effettuate PRIMA degli esiti.
      var calGood=cc.n>=300&&cg.n>=600 && cc.model/cc.n + .00001 < cc.base/cc.n && cg.model/cg.n + .00001 < cg.base/cg.n;
      // Se la calibrazione non migliora, ricade sul riferimento neutro.
      var used=calGood?pl:p0;
      // Il bound usa la dimensione effettiva della memoria locale, NON il prior come se fossero dati.
      var bound=l.n>0?confidenceLimit(l.w/l.n,Math.max(1,l.n),Z):0;
      var minGlobal=a.target>=20?5000:a.target>=5?4000:MIN_GLOBAL;
      var minLocal=a.target>=20?1100:a.target>=5?800:MIN_LOCAL;
      var enough=g.n>=minGlobal&&l.n>=minLocal;
      var riskOk=phaseAllows(a,c);
      var exp=used*a.target-1,secure=bound*a.target-1;
      var approved=riskOk&&enough&&calGood&&exp>MIN_EXPECTED&&secure>MIN_EXPECTED;
      var why=!riskOk?'RISK':!enough?'WAIT_EVIDENCE':!calGood?'CALIBRATION':!approved?'NO_EDGE':'VALIDATED';
      var x={id:a.id,family:a.family,target:a.target,p0:p0,predicted:pl,used:used,
        expected:exp,lowerEV:secure,observations:Math.round(l.n),
        globalN:Math.round(g.n),calibration:calGood,eligible:riskOk,
        approved:approved,why:why,score:secure};
      all.push(x);
      // Paper pick: misura una proposta non autorizzata; non e' una puntata.
      if(riskOk&&(!paper||exp>paper.expected))paper=x;
      if(approved&&(!best||x.score>best.score))best=x;
    }
    var reason=best?'VALIDATED':!paper?'RISK':paper.why;
    verdicts[reason]=(verdicts[reason]||0)+1;
    pending={k:k,all:all,best:best?best.id:null,paper:paper?paper.id:null,
      verdict:reason,actual:null,round:rounds+1};
    return{best:best,paperBest:paper,reason:reason,context:k,all:all};
  }
  function markActual(id,target,stake){
    if(!pending)return;
    pending.actual={id:id||'UNKNOWN',target:Number(target)||1.35,stake:Number(stake)||1};
  }
  function updateCal(z,p,p0,w){
    z.n=z.n*CAL_DECAY+1;
    z.model=z.model*CAL_DECAY+(p-w)*(p-w);
    z.base=z.base*CAL_DECAY+(p0-w)*(p0-w);
  }
  function settle(bust){
    if(!(bust>=1)||!Number.isFinite(bust))throw Error('V24: invalid bust');
    rounds++;
    if(!pending)return;
    var p=pending,loc=contexts[p.k],cal=contextCal[p.k],out={round:p.round,context:p.k,
       verdict:p.verdict,paper:p.paper||'NONE',approved:p.best||'NONE',actual:p.actual?p.actual.id:'NONE'};
    for(var i=0;i<ARMS.length;i++){
      var a=ARMS[i],was=p.all[i],won=bust>=a.target?1:0,pl=won?a.target-1:-1;
      updateCal(calibrations[i],was.predicted,was.p0,won);
      updateCal(cal[i],was.predicted,was.p0,won);
      globals[i].target=loc[i].target=a.target;
      decStat(globals[i],won);decStat(loc[i],won);
      if(was.id===p.paper){proposals++;paperPnl+=pl;out.paperPnl=pl;}
      if(was.id===p.best){authorized++;authorizedPnl+=pl;authorizedWins+=won;out.approvedPnl=pl;}
    }
    if(p.paper&&!p.best){
      rejected++;
      var paper=ARMS.find(function(a){return a.id===p.paper;});
      var win=bust>=paper.target;
      if(win)rejectedWouldWin++;else rejectedWouldLose++;
      rejectedPnl+=win?paper.target-1:-1;
    }
    if(p.actual){
      var a=p.actual,actualPnl=a.stake*(bust>=a.target?a.target-1:-1);
      executed++;executedPnl+=actualPnl;
      var f=familyActual[a.id]||(familyActual[a.id]={bets:0,wins:0,stake:0,pnl:0});
      f.bets++;f.wins+=bust>=a.target?1:0;f.stake+=a.stake;f.pnl+=actualPnl;
      if(!p.best||p.best!==a.id){reasonActual.OVERRIDE=(reasonActual.OVERRIDE||0)+1;}
      out.actualPnl=actualPnl;
    }
    if(out.actual!=='NONE'||p.best){trace.push(out);if(trace.length>300)trace.shift();}
    pending=null;
  }
  function snapshot(){
    var arms=ARMS.map(function(a,i){var g=globals[i],cal=calibrations[i];return{id:a.id,target:a.target,
        effectiveN:Math.round(g.n),virtualROI:g.n?g.pnl/g.n:0,
        brierModel:cal.n?cal.model/cal.n:null,brierBase:cal.n?cal.base/cal.n:null};});
    return{version:'2.4',rounds:rounds,contexts:Object.keys(contexts).length,
      paperChoices:proposals,validatedDecisions:authorized,validatedWins:authorizedWins,
      authorizedVirtualPnl:+authorizedPnl.toFixed(3),paperVirtualPnl:+paperPnl.toFixed(3),
      rejectedPaper:rejected,rejectedWouldWin:rejectedWouldWin,rejectedWouldLose:rejectedWouldLose,
      rejectedVirtualPnl:+rejectedPnl.toFixed(3),executed:executed,executedPnl:+executedPnl.toFixed(3),
      verdicts:verdicts,actualByArm:familyActual,arms:arms,audit:trace.slice(-10)};
  }
  function exportMemory(){
    return JSON.stringify({revision:REV,source:'observed-rounds-only',rounds:rounds,
      globals:globals,contexts:contexts,calibrations:calibrations,contextCal:contextCal});
  }
  function importMemory(json){
    var m=typeof json==='string'?JSON.parse(json):json;
    if(!m||m.revision!==REV||m.source!=='observed-rounds-only'||!Array.isArray(m.globals)||m.globals.length!==ARMS.length)throw Error('V24: memoria non compatibile');
    if(Object.keys(m.contexts||{}).length>KEY_LIMIT)throw Error('V24: memoria contesti troppo grande');
    function validRow(z){return z&&Number.isFinite(z.n)&&Number.isFinite(z.w)&&Number.isFinite(z.pnl)&&z.n>=0&&z.n<=1e9&&z.w>=0&&z.w<=z.n+1e-4;}
    for(var i=0;i<ARMS.length;i++)if(!validRow(m.globals[i]))throw Error('V24: memoria compromessa');
    for(var k in m.contexts){if(!Object.prototype.hasOwnProperty.call(m.contexts,k)||m.contexts[k].length!==ARMS.length)throw Error('V24: contesto invalido');for(var j=0;j<ARMS.length;j++)if(!validRow(m.contexts[k][j]))throw Error('V24: contesto compromesso');}
    globals=m.globals;contexts=m.contexts;calibrations=m.calibrations||ARMS.map(cal0);contextCal=m.contextCal||Object.create(null);
    rounds=Number(m.rounds)||0;pending=null;
  }
  return{arms:ARMS,plan:candidates,markActual:markActual,settle:settle,stats:snapshot,
    exportMemory:exportMemory,importMemory:importMemory,hasPending:function(){return!!pending;}};
}

/* KIBER V2.5 | Numerator Pattern Scheduler.
   No future data enters plan(). Each event is learned only when its due round ends.
   Signals are NOT profit guarantees. Eval rows are separated from training rows.
   Standalone Node export plus Bustabit-compatible ES2018 syntax. */
function createKiberV25PatternEngine(opts){
  opts=opts||{};
  var TARGETS=[1.35,1.50,2,3,5,10,20], HORIZONS=[1,2,3,4,5],
      MIN_TRIG=[0,0,12,9,6], TRAIN_MIN=[0,0,160,120,90],
      EVAL_MIN=[0,0,45,35,25], Z=opts.z||3.6;
  var past=[], rounds=0, pending=[], models=Object.create(null),
      triggered=0, approvedCount=0, approvedWins=0, approvedPnl=0,
      watched=0, discoveries=0, pastSignals=[], lastPlan=null, histMax=8;
  function bucket(x){return x<1.20?'0':x<1.50?'1':x<2?'2':x<3?'3':x<5?'4':x<10?'5':x<20?'6':'7';}
  function init(){return{n:0,trainN:0,evalN:0,trainW:[0,0,0,0,0,0,0],evalW:[0,0,0,0,0,0,0],trainBuckets:[0,0,0,0,0,0,0,0],evalBuckets:[0,0,0,0,0,0,0,0],trainSum:0,evalSum:0,last:0};}
  function wilsonLower(w,n,z){
    if(n===0)return 0;
    var p=w/n,den=1+z*z/n;
    return Math.max(0,(p+z*z/(2*n)-z*Math.sqrt((p*(1-p)+z*z/(4*n))/n))/den);
  }
  function propose(entry,now){
    var m=models[entry.key]||init(), candidates=[];
    for(var ti=0;ti<TARGETS.length;ti++){
      var target=TARGETS[ti],tWR=m.trainN?m.trainW[ti]/m.trainN:0,
          vWR=m.evalN?m.evalW[ti]/m.evalN:0,br=.99/target,
          tLower=wilsonLower(m.trainW[ti],m.trainN,Z),
          vLower=wilsonLower(m.evalW[ti],m.evalN,Z),
          valid=m.trainN>=TRAIN_MIN[entry.len]&&m.evalN>=EVAL_MIN[entry.len]&&
            tLower>1/target+0.004 && vLower>1/target+0.004;
      candidates.push({target:target,trainWR:tWR,evalWR:vWR,baseline:br,lowerTrain:tLower,
        lowerEval:vLower,train:m.trainN,eval:m.evalN,
        validated:valid,conservativeEV:Math.min(tLower,vLower)*target-1});
    }
    candidates.sort(function(a,b){return b.conservativeEV-a.conservativeEV;});
    var top=-1,nTop=0;for(var bi=0;bi<8;bi++)if(m.trainBuckets[bi]>nTop){nTop=m.trainBuckets[bi];top=bi;}
    var likelyBucket=top<0?'?':String(top), likelyP=m.trainN?nTop/m.trainN:0;
    return {likelyBucket:likelyBucket,likelyPct:+(100*likelyP).toFixed(1),
      avgNext:m.trainN?+(m.trainSum/m.trainN).toFixed(3):null,pattern:entry.key,seq:entry.seq,len:entry.len,horizon:entry.h,
      originRound:entry.origin,scheduledRound:entry.due,
      seen:m.n,trigger:m.n>=MIN_TRIG[entry.len],best:candidates[0],
      validated:candidates.some(function(x){return x.validated;}),
      bestValidated:candidates.filter(function(x){return x.validated;}).sort(function(a,b){return b.conservativeEV-a.conservativeEV;})[0]||null};
  }
  function plan(){
    var soon=rounds+1, due=[],fresh=[],ranked=[];
    // A signal can be due today even if it originated 2-5 rounds earlier.
    for(var i=0;i<pending.length;i++)if(pending[i].due===soon)due.push(pending[i]);
    if(past.length>=2){
      for(var len=2;len<=4;len++){
        if(past.length<len)continue;
        var seq=past.slice(-len).join('');
        for(var h=1;h<=5;h++){
          var e={seq:seq,len:len,h:h,key:len+':'+seq+'@+'+h,origin:rounds,due:rounds+h};
          pending.push(e);
          if(h===1){due.push(e);fresh.push(e);}
        }
      }
    }
    var selected=null, bestValidated=null;
    for(var j=0;j<due.length;j++){
      var p=propose(due[j]);
      if(p.trigger){triggered++;ranked.push(p);}
      if(p.trigger&&p.bestValidated&&(!bestValidated||p.bestValidated.conservativeEV>bestValidated.bestValidated.conservativeEV))bestValidated=p;
      if(p.trigger&&(!selected||p.seen>selected.seen))selected=p;
    }
    lastPlan={round:soon,matched:ranked.length,pattern:selected,selected:bestValidated,all:ranked,
      rawSignals:due.length,fromHistory:past.slice(-4).join('')};
    if(selected){watched++;pastSignals.push({round:soon,pattern:selected.pattern,h:selected.horizon,seen:selected.seen,approved:!!bestValidated});if(pastSignals.length>200)pastSignals.shift();}
    return lastPlan;
  }
  function settle(bust){
    if(!Number.isFinite(bust)||bust<1)throw Error('K25 pattern invalid bust');
    rounds++;
    var remaining=[],evalRow=(Math.floor((rounds-1)/500)%5)===4;
    for(var j=0;j<pending.length;j++){
      var e=pending[j];
      if(e.due>rounds){remaining.push(e);continue;}
      if(e.due!==rounds)throw Error('K25 scheduler out of sync: '+e.due+' vs '+rounds);
      var m=models[e.key];if(!m){m=models[e.key]=init();discoveries++;}
      m.n++;m.last=rounds;
      if(evalRow){m.evalN++;m.evalBuckets[Number(bucket(bust))]++;m.evalSum+=Math.min(bust,100);}
      else{m.trainN++;m.trainBuckets[Number(bucket(bust))]++;m.trainSum+=Math.min(bust,100);}
      for(var t=0;t<TARGETS.length;t++)if(bust>=TARGETS[t]){if(evalRow)m.evalW[t]++;else m.trainW[t]++;}
    }
    pending=remaining;
    if(lastPlan&&lastPlan.selected){
      var p=lastPlan.selected, target=p.bestValidated.target;
      approvedCount++;
      if(bust>=target){approvedWins++;approvedPnl+=target-1;}else approvedPnl-=1;
    }
    past.push(bucket(bust));if(past.length>histMax)past.shift();
    return lastPlan;
  }
  function stats(){return{version:'2.5',rounds:rounds,discovered:discoveries,
      patternActivations:triggered,roundsWithPattern:watched,
      validatedShadowChoices:approvedCount,validatedShadowWins:approvedWins,
      validatedShadowPnl:+approvedPnl.toFixed(3),pending:pending.length,
      last:lastPlan?{round:lastPlan.round,matched:lastPlan.matched,
        pattern:lastPlan.pattern&&lastPlan.pattern.pattern,
        horizon:lastPlan.pattern&&lastPlan.pattern.horizon,
        selected:lastPlan.selected&&lastPlan.selected.pattern,
        target:lastPlan.selected&&lastPlan.selected.bestValidated.target}:null,
      recent:pastSignals.slice(-10)};}
  function exportMemory(){
    // Da eseguire dopo GAME_ENDED. Per un uso continuativo serve salvarlo fuori da Bustabit.
    return JSON.stringify({revision:25,source:'observed-rounds-only',rounds:rounds,
      past:past,pending:pending,models:models,
      counters:{triggered:triggered,approvedCount:approvedCount,approvedWins:approvedWins,approvedPnl:approvedPnl,
        watched:watched,discoveries:discoveries,pastSignals:pastSignals}});
  }
  function importMemory(json){
    var v=typeof json==='string'?JSON.parse(json):json;
    if(!v||v.revision!==25||v.source!=='observed-rounds-only'||
      !Number.isSafeInteger(v.rounds)||v.rounds<0||!Array.isArray(v.past)||
      v.past.length>8||!Array.isArray(v.pending)||v.pending.length>100||
      !v.models||Object.keys(v.models).length>30000)throw Error('K25: memoria incompatibile');
    if(v.past.some(function(d){return !/^[0-7]$/.test(d);}))throw Error('K25: storico invalido');
    for(var i=0;i<v.pending.length;i++){
      var e=v.pending[i];
      if(!e||!Number.isInteger(e.due)||e.due<=v.rounds||e.due>v.rounds+5||
        !/^[2-4]:[0-7]{2,4}@\+[1-5]$/.test(e.key))throw Error('K25: coda compromessa');
    }
    var m=Object.create(null),names=Object.keys(v.models);
    for(var j=0;j<names.length;j++){
      var k=names[j],r=v.models[k];
      if(!/^[2-4]:[0-7]{2,4}@\+[1-5]$/.test(k)||!r||
        !Number.isFinite(r.n)||r.n<0||r.n>1e8||
        !Number.isFinite(r.trainN)||!Number.isFinite(r.evalN)||
        !Array.isArray(r.trainW)||r.trainW.length!==TARGETS.length||
        !Array.isArray(r.evalW)||r.evalW.length!==TARGETS.length||
        !Array.isArray(r.trainBuckets)||r.trainBuckets.length!==8||
        !Array.isArray(r.evalBuckets)||r.evalBuckets.length!==8)
          throw Error('K25: catalogo compromesso');
      m[k]=r;
    }
    models=m;rounds=v.rounds;past=v.past.slice();pending=v.pending;
    var c=v.counters||{};
    if(c.triggered!==undefined){
      for(var name of ['triggered','approvedCount','approvedWins','approvedPnl','watched','discoveries'])
        if(!Number.isFinite(c[name])||(name!=='approvedPnl'&&c[name]<0))throw Error('K25: contatori compromessi');
      triggered=c.triggered;approvedCount=c.approvedCount;approvedWins=c.approvedWins;
      approvedPnl=c.approvedPnl;watched=c.watched;discoveries=c.discoveries;
      pastSignals=Array.isArray(c.pastSignals)?c.pastSignals.slice(-200):[];
    }else discoveries=Object.keys(m).length;
    lastPlan=null;
  }
  return{plan:plan,settle:settle,stats:stats,
    exportMemory:exportMemory,importMemory:importMemory,
    bucket:bucket,targets:TARGETS,
    decode:{'0':'1.00-1.19x','1':'1.20-1.49x','2':'1.50-1.99x','3':'2.00-2.99x',
      '4':'3.00-4.99x','5':'5.00-9.99x','6':'10.00-19.99x','7':'>=20.00x'}};
}

/* ==============================================================
  KIBER STRIKER V2.6 - Diagnostica causale operativa (non predizione)
  Per ogni contesto registra gli esiti virtuali di tutte le opzioni.
  Le correzioni automatiche sono soglie, pause e ritorno in osservazione.
  Nessuna variazione di codice, nessuna rincorsa delle perdite.
  ============================================================== */
function createKiberV26Governor(){
  var ARMS=[['BASE_135','BASE',1.35],['BASE_150','BASE',1.5],
    ['ATTACK_200','ATTACK',2],['ATTACK_300','ATTACK',3],
    ['SENTINEL_5','SENTINEL',5],['SENTINEL_10','SENTINEL',10],
    ['SENTINEL_20','SENTINEL',20],['SENTINEL_50','SENTINEL',50],
    ['SENTINEL_100','SENTINEL',100],['RECOVERY_135','RECOVERY',1.35]];
  var stats=Object.create(null),pending=null,rounds=0;
  var diag={liveDecisions:0,skips:0,executed:0,wrongTarget:0,missedPositive:0,
    avoidedLoss:0,losses:0,wins:0,suspensions:0,swaps:0,lastReasons:[]};
  var lastAction='OBSERVE';
  function init(){return{n:0,w:0,pnl:0,last:[],lossRun:0,skipTill:0,actualN:0,actualW:0,actualPnl:0};}
  function ctx(c){
    return(c.phase||'MID')+'|'+((c.med5||0)>=1.75?'H':'L')+
      ((c.med15||0)>=1.75?'H':'L')+((c.med50||0)>=1.75?'H':'L');
  }
  function lookup(k){return stats[k]||(stats[k]=init());}
  function family(mode){if(!mode)return'NONE';if(mode.indexOf('RECOVERY')===0)return'RECOVERY';
    if(mode==='HUNT')return'SENTINEL';return mode;}
  function arm(mode,payout){
    var f=family(mode),best=null,dist=1e10;
    if(f==='CLASSIC21')return 'CLASSIC_'+String(K27_ACTIVE_CLASSIC_ID||'GEN').replace('-','_');
    if(f==='DEEP_STRIKE')return 'DEEP_'+Math.round(payout);
    if(f==='DEEP_STRIKE')return 'DEEP_'+Math.round(payout);
    for(var i=0;i<ARMS.length;i++)if(ARMS[i][1]===f){
      var d=Math.abs(ARMS[i][2]-payout);if(d<dist){dist=d;best=ARMS[i][0];}
    }
    return best||f+'_CUSTOM';
  }
  function plan(plan,context){
    if(pending)throw Error('V26: round precedente senza settlement');
    var c=ctx(context),options=(plan&&plan.all)||[];
    pending={c:c,round:rounds+1,proposals:options.map(function(a){
      return{id:a.id,family:a.family,target:a.target,approved:!!a.approved,
        reason:a.why,score:a.score};}),actual:null,
      choice:'OBSERVE',reason:(plan&&plan.reason)||'WARMUP'};
    return pending;
  }
  function gate(e,plan){
    if(!pending)return e;
    if(!e||!e.enter){diag.skips++;pending.reason=e&&e.reason||'WAIT';return e;}
    var id=arm(e.mode,e.payout),key=pending.c+'|'+id,rec=lookup(key);
    var recent=rec.last.slice(-30),sum=recent.reduce(function(a,b){return a+b;},0);
    // Recentemente persistenti insuccessi: sospensione locale, non tutte le strategie.
    if(rec.skipTill<=rounds && rec.actualN>=4 &&
       rec.actualN>(rec.lastSuspendedActual||0) &&
       ((rec.actualLossRun||0)>=4 || (recent.length>=25 && sum/recent.length<-.22 && rec.actualPnl<0))){
      rec.skipTill=rounds+45;rec.lastSuspendedActual=rec.actualN;diag.suspensions++;
    }
    if(rec.skipTill>rounds){
      diag.skips++;pending.reason='V26 sospensione '+id+' fino round '+rec.skipTill;
      return{enter:false,reason:pending.reason};
    }
    // L'autocorrezione non può promuovere un ingresso escluso dalla V2.4.
    // In CAUTION riduce l'esposizione, mantenendo sempre il tetto della V2.2.
    if(e.amount>100 && rec.n>=10 && rec.actualPnl<0){e.amount=100;
      e.reason=(e.reason||'')+' | V26 riduzione stake per perdita contestuale';}
    pending.choice=id;diag.liveDecisions++;
    return e;
  }
  function markActual(mode,target,stake){
    if(pending)pending.actual={id:arm(mode,target),target:target,stake:stake,
      family:family(mode)};
  }
  function recordReason(txt){diag.lastReasons.push(txt);if(diag.lastReasons.length>20)diag.lastReasons.shift();}
  function settle(bust){
    if(!Number.isFinite(bust)||bust<1)throw Error('V26 invalid bust');
    rounds++;
    if(!pending)return;
    var p=pending,altBest=null;
    for(var i=0;i<p.proposals.length;i++){

    classic:classic&&classic.best?Object.assign({},classic.best):null,
    deep:deep&&deep.best?Object.assign({},deep.best):null,
    dynamic:dynamic&&dynamic.selected&&dynamic.selected.bestValidated?Object.assign({},dynamic.selected.bestValidated):null,
    chosen:null,actual:null};
  for(var a of full){
    var z=armRow(a.id);z.evaluated++;
    if(a.eligible)z.eligible++;
    if(a.approved)z.approved++;
    var why=String(a.why||'UNKNOWN');z.reason[why]=(z.reason[why]||0)+1;
  }
  proposals += full.length+[pending.classic,pending.deep,pending.dynamic].filter(Boolean).length;
 }
 function virtualCandidates(){
  if(!pending)return [];
  var p=pending,a=[];
  for(var z of p.brainAll){
    a.push({mode:z.family==='SENTINEL'?'HUNT':z.family,target:z.target,
      score:z.score,reason:z.id,from:'brain',armId:z.id,
      mlApproved:!!z.approved,mlEligible:!!z.eligible,mlWhy:z.why});
  }
  if(p.classic)a.push({mode:'CLASSIC21',target:p.classic.target,score:p.classic.score,reason:p.classic.id,from:'classic',patternId:p.classic.id});
  if(p.deep)a.push({mode:'DEEP_STRIKE',target:p.deep.target,score:p.deep.score,reason:p.deep.id,from:'deep'});
  if(p.dynamic)a.push({mode:'PATTERN',target:p.dynamic.target,score:p.dynamic.conservativeEV,reason:'V25',from:'pattern'});
  return a;
 }
 function vetoCandidate(c,reason){var k=c.mode+':'+reason;candidateVeto[k]=(candidateVeto[k]||0)+1;}
 function safety(c,ctx){
  if(c.mode==='DEEP_STRIKE'){
   // Deep uses its OWN conservative out-of-sample validation, not generic ATTACK
   // confidence/targetPotential. High-tail wagers remain one-bit and SHADOW-only.
   if(STRIKER_LIVE)return false;
   if(ctx.phase==='NEGATIVE'||ctx.givebackBrake===2)return false;
   if(c.from!=='deep'||!Number.isFinite(c.score)||c.score<=.5)return false;
  }
  if(c.mode==='CLASSIC21' && (ctx.givebackBrake===2||!config.enableClassic21.value))return false;
  if(c.mode==='PATTERN' && (ctx.givebackBrake===2||!config.mlSupervisor.value))return false;
  if(c.mode==='DEEP_STRIKE'&&!config.enableDeepStrike.value)return false;
  if(c.mode==='SENTINEL'||c.mode==='HUNT'){
   // Study/SHADOW only unless all separate authorizations allow real wager.
   if(STRIKER_LIVE&&(!config.huntLive.value||!config.mlSentinelLive.value))return false;
  }
  if(c.mode==='ATTACK'&&(ctx.attackBlocked||ctx.phase==='NEGATIVE'||ctx.confidence<73))return false;
  return true;
 }
 function isStale(c){
  var v=perf(pkey(c.mode,c.target,pending.phase));
  if(v.suspendedUntil>rounds){rejectedByActual++;return true;}
  // Actual selections alone may be biased by the old arbiter: require sustained losses,
  // not a single missed 10x. Reconsider after cooling off; shadow statistics keep training.
  if(v.actualLast.length>=4 && v.lossRun>=4 && v.actualN>(v.lastSuspendedActualN||0)){
    v.suspendedUntil=rounds+60;v.lastSuspendedActualN=v.actualN;v.reason='four successive realized selection losses';
    rejectedByActual++;return true;
  }
  var rec=v.actualLast.slice(-16),pl=rec.reduce(function(a,b){return a+b;},0);
  if(rec.length>=12&&pl/rec.length<-.30&&v.actualN>(v.lastSuspendedActualN||0)){
    v.suspendedUntil=rounds+100;v.lastSuspendedActualN=v.actualN;v.reason='negative rolling actual return';
    rejectedByActual++;return true;
  }
  var sim=v.virtualLast.slice(-40),pnl=sim.reduce(function(a,b){return a+b;},0);
  if(sim.length>=35&&pnl/sim.length<-.25){
    rejectedByVirtual++;return true;
  }
  return false;
 }
 function choose(e,capState,context){
  context=context||{};
  if(!pending)return e;
  if(['HARD_STOP','OBSERVE','PROFIT_LOCK','PROBE'].indexOf(capState)>=0)return e;
  var cand=virtualCandidates(), choices=[];
  for(var x of cand){if(x.from==='classic'&&(!config.enableClassic21.value))continue;
   if(x.from==='deep'&&(!config.enableDeepStrike.value))continue;
   if(!safety(x,context)){vetoCandidate(x,'SPECIALIST_OR_PHASE_RISK');continue;}
   if(x.from==='brain'){
     // V2.9: independent approval for EACH target, not just plan.best.
     // An unapproved arm must NEVER become a wager, even if it has a high score.
     if(!config.mlSupervisor.value||!x.mlApproved||!x.mlEligible){vetoCandidate(x,'ML_NOT_APPROVED_OR_INELIGIBLE');continue;}
     // Preserve the separate Sentinel real-money double confirmation.
     if(x.mode==='HUNT'&&STRIKER_LIVE){
       var hh=ksiChooseHunt(buildSnapshot());
       if(!hh||!ksiHuntMayGoLive(hh)||Math.abs(hh.target-x.target)>.001)continue;
     }
     x.amount=100;x.payout=x.target;
     if(x.armId)armRow(x.armId).considered++;
   }else if(x.from==='pattern'){
     if(STRIKER_LIVE&&!config.patternLive.value)continue;
     x.amount=100;
   }else x.amount=100;
   if(isStale(x)){vetoCandidate(x,'PERFORMANCE_SUSPENSION');continue;}
   // V29 FIX: the old -.1..+.25 clamp made every profitable arm score exactly .25;
   // the additional target penalty then ALWAYS favoured 1.35x over a superior,
   // independently validated ATTACK 2x/3x. Preserve ranking information.
   // Conservative expected return is comparable across arms; compress rather
   // than truncate, then penalize longer-tailed targets for their dispersion.
   var raw=Math.max(-1,Math.min(20,Number(x.score)||0));
   var score=(raw>=0?1:-1)*Math.log1p(Math.abs(raw));
   // DEEP has already paid a stringent train/eval Wilson lower-bound penalty.
   // Avoid penalizing its high target AGAIN as if it lacked that evidence.
   score=score/(1+Math.log2(Math.max(1.01,x.target))*(x.from==='deep'?.08:.22));
   var v=perf(pkey(x.mode,x.target,pending.phase));
   if(v.actualN>=8)score-=Math.max(0,-v.actualPnl/v.actualN)*.15;
   if(v.actualN>0&&v.lossRun>=2)score-=Math.min(.12,.03*v.lossRun);
   choices.push({x:x,score:score});
  }
  // Legacy selection might be PATTERN from ML and already be included above. If a valid
  // legacy recovery/non-model choice exists, retain it unless explicit safety states forbid it.
  if(!choices.length){
    noCandidate++;
    if(e&&e.enter){
      var k=pkey(e.mode,e.payout,pending.phase);
      if((e.mode==='PATTERN'||e.mode==='CLASSIC21'||e.mode==='DEEP_STRIKE') && isStale({mode:e.mode,target:e.payout}))
        return {enter:false,reason:'V28: chosen strategy under performance suspension'};
      if(e.mode==='PATTERN')return{enter:false,reason:'V28: pattern lacks currently eligible counterfactual candidate'};
      // A legacy BASE/ATTACK/HUNT proposal is not an exception to ML validation.
      if(e.mode==='HUNT'||e.mode==='ATTACK'||e.mode==='BASE')
        return {enter:false,reason:'V29: brain arm not independently ML-approved / risk eligible'};
      return e;
    }
    return {enter:false,reason:'V28: none of the strategies currently satisfies ML and risk criteria'};
  }
  // Stable, deterministic tie breaks; candidate order must not determine the winner.
  choices.sort(function(a,b){return b.score-a.score||String(a.x.mode).localeCompare(String(b.x.mode))||a.x.target-b.x.target;});
  var x=choices[0].x;
  pending.chosen={mode:x.mode,target:x.target,score:choices[0].score,armId:x.armId||null};
  if(x.from==='brain'&&x.armId)armRow(x.armId).selected++;
  if(x.from==='classic')K27_ACTIVE_CLASSIC_ID=x.patternId;
  if(e&&e.enter&&family(e.mode)===family(x.mode)&&Math.abs(e.payout-x.target)<.001)return e;
  return{enter:true,mode:x.mode,amount:x.amount||100,payout:x.target,
   reason:'V28 ARBITER '+x.from+' '+x.reason+' score='+choices[0].score.toFixed(4)};
 }
 function noteVeto(reason){
   if(!pending||!pending.chosen||pending.actual)return;
   var raw=String(reason||'UNCLASSIFIED');
   var k=raw;
   if(raw.indexOf('V30_DRIFT_REVALIDATION')>=0)k='DRIFT_PAUSE';
   else if(raw.indexOf('V26 sospensione')>=0){
     var id=(raw.match(/V26 sospensione ([A-Z_0-9]+)/)||[])[1]||'UNKNOWN';
     k='GOVERNOR_SUSPENSION:'+id;
   }
   k=k.slice(0,90);
   abortReasons[k]=(abortReasons[k]||0)+1;pending.abortReason=k;
   abortSamples.push({round:pending.round,reason:raw.slice(0,130)});
   if(abortSamples.length>30)abortSamples.shift();
 }
 function markDirect(mode,target,reason){
   if(pending)pending.chosen={mode:mode,target:target,score:null,reason:reason||'DIRECT'};
 }
 function size(e,balance,capState){
  if(!e||!e.enter||!config.enableDynamicSizing.value)return e;
  if(capState!=='NORMAL'||e.mode.indexOf('RECOVERY')===0||['HUNT','DEEP_STRIKE','CLASSIC21','PATTERN'].includes(e.mode))return e;
  var r=row(family(e.mode)),roi=r.stake?r.pnl/r.stake:0;
  if(r.actualN<15||roi<=.04||r.actualPnl<=0)return e;
  var extra=Math.min(8,1+Math.floor(Math.max(0,r.actualN-15)/15),1+Math.floor(Math.max(0,roi-.04)*15));
  var afford=Math.max(1,Math.floor((balance/100)*.0025));
  var left=cycleActive?Math.floor(Math.max(0,cycleBudget-cycleExposure)/100):8;
  var bitsN=Math.max(1,Math.min(extra,afford,left,8));
  e.amount=Math.min(bitsN*100,Math.max(100,Number(config.maxBetBits.value||8)*100));
  e.reason=(e.reason||'')+' | V28 sizing '+bitsN+' bit';return e;
 }
 function markActual(mode,target,stake){if(pending)pending.actual={mode:family(mode),target:target,stake:stake,phase:pending.phase};}
 function settle(bust){
  rounds++;if(!pending)return;
  var p=pending,candidates=virtualCandidates();
  for(var c of candidates){var m=family(c.mode),r=row(m),win=bust>=c.target,pl=win?c.target-1:-1;
   r.n++;r.w+=win?1:0;r.pnl+=pl;r.stake++;r.last.push(pl);if(r.last.length>50)r.last.shift();
   var v=perf(pkey(c.mode,c.target,p.phase));v.virtualN++;v.virtualPnl+=pl;v.virtualLast.push(pl);if(v.virtualLast.length>50)v.virtualLast.shift();
   if(c.from==='brain'&&c.armId){
     var az=armRow(c.armId);az.shadowPnl+=pl;if(win)az.shadowWins++;
   }
  }
  if(p.actual){var a=p.actual,r=row(a.mode),won=bust>=a.target,pl=a.stake*(won?a.target-1:-1);
   r.actualN++;r.actualW+=won?1:0;r.actualPnl+=pl;executed++;
   var v=perf(pkey(a.mode,a.target,a.phase));v.actualN++;v.actualW+=won?1:0;v.actualPnl+=pl;
   v.actualLast.push(pl/a.stake);if(v.actualLast.length>50)v.actualLast.shift();
   v.lossRun=won?0:v.lossRun+1;
   if(p.chosen&&p.chosen.armId)armRow(p.chosen.armId).executed++;
  }
  if(p.chosen){
   selected++;
   if(!p.actual && !p.abortReason){
     abortReasons.UNCLASSIFIED=(abortReasons.UNCLASSIFIED||0)+1;
   }
  }
  pending=null;
 }
 function report(){return{version:'2.9',rounds:rounds,proposals:proposals,selected:selected,executed:executed,
   rejectedActual:rejectedByActual,rejectedVirtual:rejectedByVirtual,noCandidate:noCandidate,stats:rows,
   performance:performance,brainArms:armAudit,candidateVeto:candidateVeto,postSelectionVeto:abortReasons,recentVetoSamples:abortSamples.slice(-10)};}
 function exportMemory(){return{revision:27,rounds:rounds,proposals:proposals,selected:selected,
    actual:executed,stats:rows,performance:performance,brainArms:armAudit,
    rejectedActual:rejectedByActual,rejectedVirtual:rejectedByVirtual,noCandidate:noCandidate,
    candidateVeto:candidateVeto,postSelectionVeto:abortReasons,recentVetoSamples:abortSamples};}
 function importMemory(v){if(!v||v.revision!==27||!Number.isSafeInteger(v.rounds)||!v.stats)throw Error('K29 ENSEMBLE memory invalid');
  rounds=v.rounds;proposals=v.proposals||0;selected=v.selected||0;executed=v.actual||0;
  rows=Object.assign(Object.create(null),v.stats);performance=Object.assign(Object.create(null),v.performance||{});
  armAudit=Object.assign(Object.create(null),v.brainArms||{});
  rejectedByActual=v.rejectedActual||0;rejectedByVirtual=v.rejectedVirtual||0;noCandidate=v.noCandidate||0;
  candidateVeto=Object.assign(Object.create(null),v.candidateVeto||{});
  abortReasons=Object.assign(Object.create(null),v.postSelectionVeto||{});
  abortSamples=Array.isArray(v.recentVetoSamples)?v.recentVetoSamples.slice(-30):[];pending=null;
 }
 return{begin:begin,choose:choose,size:size,markActual:markActual,markDirect:markDirect,noteVeto:noteVeto,settle:settle,report:report,exportMemory:exportMemory,importMemory:importMemory};
}


/* V3.0: conservative regime-change adaptation. Only completed real/SHADOW
   decisions are used to update the state. No martingale, no wager increases.
   A selected-but-vetoed proposal remains an ex-ante paper prediction. */
function createKiberV30Drift(){
  var rounds=0,pending=null,actual=[],paper=[],pauseUntil=0;
  var triggered=0,paused=0,resumed=0,extended=0;
  function avg(a){return a.length?a.reduce(function(s,x){return s+x;},0)/a.length:0;}
  function plan(e){pending=e&&e.enter&&Number.isFinite(e.payout)?{
    target:e.payout,stake:1,planned:rounds+1,actual:false}:null;}
  function gate(e,state){
    if(!e||!e.enter||state!=='NORMAL'||STRIKER_LIVE)return e;
    if(actual.length>=20 && avg(actual.slice(-20))<-.24 && rounds>=pauseUntil){
      pauseUntil=rounds+60;triggered++;actual=[];
    }
    if(pauseUntil>0 && rounds>=pauseUntil){
      var recent=paper.slice(-45);
      if(recent.length>=25 && avg(recent)<=0){
        pauseUntil=rounds+40;extended++;
      }else{pauseUntil=0;resumed++;}
    }
    if(rounds<pauseUntil){paused++;
      return{enter:false,reason:'V30_DRIFT_REVALIDATION until '+pauseUntil};
    }
    return e;
  }
  function markActual(target,stake){
    if(pending){pending.actual=true;pending.target=target;pending.stake=stake;}
  }
  function settle(bust){
    rounds++;
    if(pending){
      var v=bust>=pending.target?pending.target-1:-1;
      paper.push(v);if(paper.length>60)paper.shift();
      if(pending.actual){actual.push(v);if(actual.length>40)actual.shift();}
    }
    pending=null;
  }
  function stats(){return{rounds:rounds,triggered:triggered,pausedRounds:paused,resumed:resumed,
    extended:extended,pauseUntil:pauseUntil,recentActual:actual.length,recentPaper:paper.length,
    actualMean:avg(actual.slice(-20)),paperMean:avg(paper.slice(-45))};}
  function exportMemory(){return{revision:30,rounds:rounds,actual:actual,paper:paper,
    pauseUntil:pauseUntil,triggered:triggered,paused:paused,resumed:resumed,extended:extended};}
  function importMemory(m){if(!m||m.revision!==30)return;
    if(!Number.isSafeInteger(m.rounds)||!Array.isArray(m.actual)||!Array.isArray(m.paper)||
        m.actual.length>40||m.paper.length>60)throw Error('V30 invalid drift memory');
    rounds=m.rounds;actual=m.actual;paper=m.paper;pauseUntil=Number(m.pauseUntil)||0;
    triggered=Number(m.triggered)||0;paused=Number(m.paused)||0;
    resumed=Number(m.resumed)||0;extended=Number(m.extended)||0;pending=null;
  }
  return{plan:plan,gate:gate,markActual:markActual,settle:settle,
    stats:stats,exportMemory:exportMemory,importMemory:importMemory};
}
