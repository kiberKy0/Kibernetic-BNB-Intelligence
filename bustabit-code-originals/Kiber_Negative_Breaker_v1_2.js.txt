var config = {
  targetProfit: { value: 150, type: 'balance', label: 'Profitto minimo per colpo (bits)' },
  maxBet: { value: 1000, type: 'balance', label: 'Puntata massima recovery (bits)' },
  playRate: { value: '65', type: 'text', label: 'Game giocati ogni 100' },
  reserveTrigger: { value: 300, type: 'balance', label: 'Riserva minima Moonshot (bits)' },
  reserveCut: { value: '20', type: 'text', label: '% profitto messo in riserva' },
  moonMin: { value: '80', type: 'text', label: 'Moonshot payout minimo' },
  moonMax: { value: '500', type: 'text', label: 'Moonshot payout massimo' },
  moonMinBet: { value: 100, type: 'balance', label: 'Puntata minima Moonshot (bits)' },
  moonMaxBet: { value: 300, type: 'balance', label: 'Puntata massima Moonshot (bits)' },
  maxBankrollPct: { value: '8', type: 'text', label: 'Max % bankroll per singolo colpo' },
  maxMoonDebt: { value: 150, type: 'balance', label: 'Debito massimo per consentire Moonshot (bits)' }
};

var TARGET_BITS = config.targetProfit.value / 100;
var MAX_BET_BITS = config.maxBet.value / 100;
var PLAY_TARGET = clampInt(parseInt(config.playRate.value, 10) || 65, 1, 100);
var RESERVE_TRIGGER = config.reserveTrigger.value / 100;
var RESERVE_CUT = clamp((parseFloat(config.reserveCut.value) || 20) / 100, 0, 0.75);
var MOON_MIN = clamp(parseFloat(config.moonMin.value) || 80, 20, 1000);
var MOON_MAX = clamp(parseFloat(config.moonMax.value) || 500, MOON_MIN, 5000);
var MOON_MIN_BET_BITS = config.moonMinBet.value / 100;
var MOON_MAX_BET_BITS = config.moonMaxBet.value / 100;
var MAX_BANKROLL_PCT = clamp((parseFloat(config.maxBankrollPct.value) || 8) / 100, 0.005, 0.50);
var MAX_MOON_DEBT_BITS = config.maxMoonDebt.value / 100;

var windowGames = 0;
var windowPlayed = 0;
var totalPlayed = 0;
var totalWon = 0;
var totalLost = 0;
var sessionPnl = 0;
var reserveBits = 0;
var recoveryDebt = 0;
var winStreak = 0;
var lossStreak = 0;
var lastMode = 'NONE';
var lastProcessedGameId = null;
var moonCooldown = 0;

function clamp(v, lo, hi) {
  return Math.max(lo, Math.min(hi, v));
}

function clampInt(v, lo, hi) {
  return Math.floor(clamp(v, lo, hi));
}

function bitsToSat(bits) {
  var sat = Math.round(Number(bits) * 100);
  if (!isFinite(sat)) return 0;
  return Math.floor(sat);
}

function safeBetSat(bits, minSat) {
  var sat = bitsToSat(bits);
  var minimum = Math.max(1, Math.floor(minSat || 1));
  if (sat < minimum) sat = minimum;
  return Math.floor(sat);
}

function satToBits(sat) {
  return Number(sat) / 100;
}

function r2(v) {
  return Math.round(Number(v) * 100) / 100;
}

function rnd(min, max) {
  return min + Math.random() * (max - min);
}

function completedHistory() {
  var arr = engine.history.toArray().filter(function (g) {
    return typeof g.bust === 'number' && isFinite(g.bust);
  });

  arr.sort(function (a, b) {
    return (b.id || 0) - (a.id || 0);
  });

  return arr;
}

function recent(arr, n) {
  return arr.slice(0, Math.min(n, arr.length));
}

function countWhere(arr, fn) {
  var c = 0;
  var i;
  for (i = 0; i < arr.length; i++) {
    if (fn(arr[i])) c++;
  }
  return c;
}

function redStreak(hist) {
  var s = 0;
  var i;
  for (i = 0; i < hist.length; i++) {
    if (hist[i].bust < 2) s++;
    else break;
  }
  return s;
}

function analyzeMoment() {
  var hist = completedHistory();
  var h8 = recent(hist, 8);
  var h12 = recent(hist, 12);
  var h20 = recent(hist, 20);

  var reds8 = countWhere(h8, function (g) { return g.bust < 2; });
  var reds12 = countWhere(h12, function (g) { return g.bust < 2; });
  var reds20 = countWhere(h20, function (g) { return g.bust < 2; });
  var deep8 = countWhere(h8, function (g) { return g.bust < 1.30; });
  var greens3 = countWhere(h12, function (g) { return g.bust >= 3; });
  var highs10 = countWhere(h20, function (g) { return g.bust >= 10; });
  var rs = redStreak(hist);
  var redRate20 = h20.length ? reds20 / h20.length : 0.5;

  var blockDanger =
    (h12.length >= 8 && reds12 >= 8 && deep8 >= 3) ||
    rs >= 7;

  var score = 0;
  if (rs >= 3) score += 2;
  if (rs >= 5) score += 1;
  if (redRate20 >= 0.60) score += 1;
  if (greens3 >= 4) score += 1;
  if (highs10 >= 2) score -= 0.5;
  if (blockDanger) score -= 2;

  return {
    redStreak: rs,
    reds8: reds8,
    reds12: reds12,
    redRate20: redRate20,
    deep8: deep8,
    greens3: greens3,
    highs10: highs10,
    blockDanger: blockDanger,
    score: score
  };
}

function shouldPlay(moment) {
  var remainingIncludingThis = 101 - windowGames;
  var need = PLAY_TARGET - windowPlayed;

  if (windowPlayed >= PLAY_TARGET) return false;
  if (need >= remainingIncludingThis) return true;

  var p = PLAY_TARGET / 100;

  if (moment.redStreak >= 3) p += 0.18;
  if (moment.redStreak >= 5) p += 0.07;
  if (moment.blockDanger) p -= 0.22;
  if (recoveryDebt > 0) p += 0.07;

  p = clamp(p, 0.20, 0.92);
  return Math.random() < p;
}

function chooseRegularPayout(moment) {
  var payout = 1.62;

  if (moment.redStreak >= 3) payout = 1.55;
  if (moment.redStreak >= 5) payout = 1.48;
  if (moment.redStreak >= 7) payout = 1.42;

  if (sessionPnl > 0) {
    payout += Math.min(0.35, winStreak * 0.08);
  }

  if (sessionPnl > TARGET_BITS * 6) payout += 0.10;

  if (winStreak >= 4 && !moment.blockDanger) {
    payout += 0.20;
  }

  if (recoveryDebt > 0) {
    payout = Math.min(payout, 1.75);
  }

  if (moment.blockDanger) {
    payout = Math.min(payout, 1.50);
  }

  payout += rnd(-0.04, 0.05);

  return r2(clamp(payout, 1.35, 2.50));
}

function recoveryTarget() {
  if (recoveryDebt <= 0) return TARGET_BITS;

  var slice = Math.min(
    4.0,
    Math.max(0.35, recoveryDebt * 0.28)
  );

  return TARGET_BITS + slice;
}

function bankrollCapBits() {
  var balBits = satToBits(userInfo.balance);
  return Math.max(0.01, balBits * MAX_BANKROLL_PCT);
}

function regularPlan(moment) {
  var payout = chooseRegularPayout(moment);
  var desiredProfit = recoveryTarget();
  var stake = desiredProfit / (payout - 1);

  if (moment.redStreak >= 4 && !moment.blockDanger) {
    stake *= 1.10;
  }

  if (lossStreak >= 2) {
    stake *= Math.min(1.35, 1 + lossStreak * 0.06);
  }

  var cap = Math.min(MAX_BET_BITS, bankrollCapBits());
  stake = clamp(stake, 0.01, cap);

  if (stake * (payout - 1) < desiredProfit) {
    var neededPayout = 1 + (desiredProfit / stake);
    payout = r2(clamp(Math.max(payout, neededPayout), 1.35, 4.50));
  }

  var plannedProfit = stake * (payout - 1);

  return {
    mode: recoveryDebt > 0 ? 'RECOVERY' : (moment.redStreak >= 3 ? 'NEG_BREAK' : 'NORMAL'),
    stakeBits: r2(stake),
    payout: payout,
    desiredProfit: r2(plannedProfit),
    minBetSat: 1
  };
}

function moonshotReady(moment) {
  if (reserveBits < RESERVE_TRIGGER) return false;
  if (moonCooldown > 0) return false;
  if (moment.blockDanger) return false;

  // FIX V1.2: niente Moonshot durante una sessione negativa.
  if (sessionPnl <= 0) return false;

  // FIX V1.2: il recupero deve essere quasi chiuso.
  if (recoveryDebt > MAX_MOON_DEBT_BITS) return false;

  var chance = 0.08;

  if (moment.redStreak >= 4) chance += 0.05;
  if (winStreak >= 2) chance += 0.04;
  if (sessionPnl > TARGET_BITS * 5) chance += 0.03;

  return Math.random() < clamp(chance, 0.05, 0.20);
}

function moonshotPlan(moment) {
  var reserveStrength = clamp(
    (reserveBits - RESERVE_TRIGGER) /
    Math.max(1, RESERVE_TRIGGER * 4),
    0,
    1
  );

  var payout = MOON_MIN + (MOON_MAX - MOON_MIN) * reserveStrength;

  if (moment.redStreak >= 5) {
    payout += rnd(20, 80);
  } else {
    payout += rnd(-10, 30);
  }

  payout = r2(clamp(payout, MOON_MIN, MOON_MAX));

  // FIX V1.2: almeno 1 bit, poi cresce con la riserva fino al massimo configurato.
  var stake = reserveBits * 0.08;
  stake = Math.max(MOON_MIN_BET_BITS, stake);
  stake = Math.min(MOON_MAX_BET_BITS, stake);
  stake = Math.min(stake, bankrollCapBits());

  return {
    mode: 'MOONSHOT',
    stakeBits: r2(stake),
    payout: payout,
    desiredProfit: r2(stake * (payout - 1)),
    minBetSat: 100
  };
}

function placePlan(plan, moment) {
  var minSat = plan.minBetSat || 1;
  var stakeSat = safeBetSat(plan.stakeBits, minSat);
  var payout = r2(Number(plan.payout));

  if (!isFinite(stakeSat) || stakeSat < minSat || Math.floor(stakeSat) !== stakeSat) {
    log('[BET ERROR] Puntata non valida: ' + stakeSat + ' sat');
    return false;
  }

  if (!isFinite(payout) || payout <= 1) {
    log('[BET ERROR] Payout non valido: ' + payout);
    return false;
  }

  if (stakeSat > userInfo.balance) {
    log('[SKIP] Saldo insufficiente per ' + r2(satToBits(stakeSat)) + ' bits');
    return false;
  }

  var actualStakeBits = satToBits(stakeSat);

  try {
    engine.bet(stakeSat, payout);
  } catch (e) {
    log('[BET ERROR] ' + (e && e.message ? e.message : String(e)) +
        ' | sat=' + stakeSat + ' | payout=' + payout);
    return false;
  }

  lastMode = plan.mode;
  windowPlayed++;
  totalPlayed++;

  if (plan.mode === 'MOONSHOT') {
    reserveBits = Math.max(0, reserveBits - actualStakeBits);
    moonCooldown = 8 + Math.floor(Math.random() * 12);
  }

  log(
    '[BET] ' + plan.mode +
    ' | ' + r2(actualStakeBits) + ' bits @ ' + payout + 'x' +
    ' | target≈+' + r2(actualStakeBits * (payout - 1)) +
    ' | redStreak=' + moment.redStreak +
    ' | debt=' + r2(recoveryDebt) +
    ' | reserve=' + r2(reserveBits) +
    ' | quota=' + windowPlayed + '/' + PLAY_TARGET
  );

  return true;
}

function processLastResult() {
  var g = engine.history.first();

  if (!g || typeof g.bust !== 'number') return;
  if (g.id === lastProcessedGameId) return;

  lastProcessedGameId = g.id;

  if (moonCooldown > 0) moonCooldown--;

  if (!g.wager || g.wager <= 0) return;

  var stakeBits = satToBits(g.wager);
  var won = typeof g.cashedAt === 'number' && g.cashedAt > 1;

  if (won) {
    var profitBits = stakeBits * (g.cashedAt - 1);

    totalWon++;
    winStreak++;
    lossStreak = 0;
    sessionPnl += profitBits;

    if (lastMode === 'MOONSHOT') {
      reserveBits += stakeBits + profitBits;
    } else {
      var reserved = profitBits * RESERVE_CUT;
      reserveBits += reserved;
      recoveryDebt = Math.max(0, recoveryDebt - profitBits * 0.60);
    }

    log(
      '[WIN] +' + r2(profitBits) +
      ' bits | P/L=' + r2(sessionPnl) +
      ' | debt=' + r2(recoveryDebt) +
      ' | reserve=' + r2(reserveBits)
    );

  } else {
    totalLost++;
    lossStreak++;
    winStreak = 0;
    sessionPnl -= stakeBits;

    if (lastMode !== 'MOONSHOT') {
      recoveryDebt += stakeBits;
    }

    log(
      '[LOSS] -' + r2(stakeBits) +
      ' bits | debt=' + r2(recoveryDebt) +
      ' | P/L=' + r2(sessionPnl) +
      ' | reserve=' + r2(reserveBits)
    );
  }
}

engine.on('GAME_ENDED', function () {
  processLastResult();
});

engine.on('GAME_STARTING', function () {
  var moment = analyzeMoment();
  windowGames++;

  if (windowGames > 100) {
    log(
      '[100 GAME]' +
      ' giocati=' + windowPlayed + '/100' +
      ' | W=' + totalWon +
      ' L=' + totalLost +
      ' | P/L=' + r2(sessionPnl) +
      ' | debt=' + r2(recoveryDebt) +
      ' | reserve=' + r2(reserveBits)
    );

    windowGames = 1;
    windowPlayed = 0;
  }

  if (!shouldPlay(moment)) {
    log(
      '[SKIP]' +
      ' quota ' + windowPlayed + '/' + PLAY_TARGET +
      ' | redStreak=' + moment.redStreak +
      (moment.blockDanger ? ' | BLOCK GUARD' : '')
    );
    return;
  }

  var plan = moonshotReady(moment)
    ? moonshotPlan(moment)
    : regularPlan(moment);

  placePlan(plan, moment);
});

log('KIBER NEGATIVE BREAKER v1.2 avviato');
log(
  'Target: ' + PLAY_TARGET + '/100 game' +
  ' | profit target: ' + TARGET_BITS + ' bits' +
  ' | reserve trigger: ' + RESERVE_TRIGGER + ' bits' +
  ' | Moonshot: ' + MOON_MIN + 'x-' + MOON_MAX + 'x' +
  ' | Moon min bet: ' + MOON_MIN_BET_BITS + ' bit'
);
