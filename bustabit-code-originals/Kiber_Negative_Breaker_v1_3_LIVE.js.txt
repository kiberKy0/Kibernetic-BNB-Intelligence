var config = {
  targetProfit: { value: 150, type: 'balance', label: 'Profitto minimo per colpo' },
  maxBet: { value: 1000, type: 'balance', label: 'Puntata massima normale' },
  defenseMaxBet: { value: 600, type: 'balance', label: 'Puntata massima DEFENSE' },
  playRate: { value: '65', type: 'text', label: 'Target massimo game ogni 100' },
  minPlayRate: { value: '50', type: 'text', label: 'Target minimo game ogni 100' },
  reserveTrigger: { value: 300, type: 'balance', label: 'Riserva minima Moonshot' },
  reserveCut: { value: '20', type: 'text', label: '% profitto in riserva' },
  moonMin: { value: '80', type: 'text', label: 'Moonshot minimo' },
  moonMax: { value: '500', type: 'text', label: 'Moonshot massimo' },
  moonMinBet: { value: 100, type: 'balance', label: 'Puntata minima Moonshot' },
  moonMaxBet: { value: 300, type: 'balance', label: 'Puntata massima Moonshot' },
  maxBankrollPct: { value: '8', type: 'text', label: 'Max % bankroll per colpo' },
  defenseDrawdown: { value: 2000, type: 'balance', label: 'Drawdown attiva DEFENSE' },
  moonMinSessionProfit: { value: 1000, type: 'balance', label: 'Profitto minimo per Moonshot' }
};

var TARGET_BITS = config.targetProfit.value / 100;
var MAX_BET_BITS = config.maxBet.value / 100;
var DEFENSE_MAX_BET_BITS = config.defenseMaxBet.value / 100;
var PLAY_TARGET = clampInt(parseInt(config.playRate.value, 10) || 65, 1, 100);
var MIN_PLAY_TARGET = clampInt(parseInt(config.minPlayRate.value, 10) || 50, 1, PLAY_TARGET);
var RESERVE_TRIGGER = config.reserveTrigger.value / 100;
var RESERVE_CUT = clamp((parseFloat(config.reserveCut.value) || 20) / 100, 0, 0.75);
var MOON_MIN = clamp(parseFloat(config.moonMin.value) || 80, 20, 1000);
var MOON_MAX = clamp(parseFloat(config.moonMax.value) || 500, MOON_MIN, 5000);
var MOON_MIN_BET_BITS = config.moonMinBet.value / 100;
var MOON_MAX_BET_BITS = config.moonMaxBet.value / 100;
var MAX_BANKROLL_PCT = clamp((parseFloat(config.maxBankrollPct.value) || 8) / 100, 0.005, 0.50);
var DEFENSE_DRAWDOWN_BITS = config.defenseDrawdown.value / 100;
var MOON_MIN_SESSION_PROFIT_BITS = config.moonMinSessionProfit.value / 100;

var windowGames = 0;
var windowPlayed = 0;
var totalPlayed = 0;
var totalWon = 0;
var totalLost = 0;
var sessionPnl = 0;
var peakPnl = 0;
var currentDrawdown = 0;
var reserveBits = 0;
var recoveryDebt = 0;
var winStreak = 0;
var lossStreak = 0;
var lastMode = 'NONE';
var lastProcessedGameId = null;
var moonCooldown = 0;

function clamp(v, lo, hi) {
  return Math.max(lo, Math.min(hi, v));
}

function clampInt(v, lo, hi) {
  return Math.floor(clamp(v, lo, hi));
}

function r2(v) {
  return Math.round(Number(v) * 100) / 100;
}

function rnd(min, max) {
  return min + Math.random() * (max - min);
}

function satToBits(sat) {
  return Number(sat) / 100;
}

/* LIVE FIX: sempre bit interi, quindi multipli di 100 satoshi. */
function bitsToSat(bits) {
  var value = Number(bits);
  if (!isFinite(value) || value <= 0) return 0;

  var wholeBits = Math.max(1, Math.ceil(value));
  return wholeBits * 100;
}

function safeBetSat(bits, minSat) {
  var sat = bitsToSat(bits);
  var minimum = Math.max(
    100,
    Math.ceil(Number(minSat || 100) / 100) * 100
  );

  if (sat < minimum) sat = minimum;

  sat = Math.ceil(sat / 100) * 100;
  return Math.floor(sat);
}

function completedHistory() {
  var arr = engine.history.toArray().filter(function (g) {
    return typeof g.bust === 'number' && isFinite(g.bust);
  });

  arr.sort(function (a, b) {
    return (b.id || 0) - (a.id || 0);
  });

  return arr;
}

function recent(arr, n) {
  return arr.slice(0, Math.min(n, arr.length));
}

function countWhere(arr, fn) {
  var c = 0;
  for (var i = 0; i < arr.length; i++) {
    if (fn(arr[i])) c++;
  }
  return c;
}

function redStreak(hist) {
  var s = 0;
  for (var i = 0; i < hist.length; i++) {
    if (hist[i].bust < 2) s++;
    else break;
  }
  return s;
}

function updateRiskState() {
  if (sessionPnl > peakPnl) peakPnl = sessionPnl;
  currentDrawdown = Math.max(0, peakPnl - sessionPnl);

  /*
   * V1.3: il recovery debt segue SOLO la perdita reale della sessione.
   * Se il P/L torna positivo, debt = 0.
   */
  recoveryDebt = Math.max(0, -sessionPnl);
}

function analyzeMoment() {
  var hist = completedHistory();
  var h8 = recent(hist, 8);
  var h12 = recent(hist, 12);
  var h20 = recent(hist, 20);

  var reds8 = countWhere(h8, function (g) { return g.bust < 2; });
  var reds12 = countWhere(h12, function (g) { return g.bust < 2; });
  var reds20 = countWhere(h20, function (g) { return g.bust < 2; });
  var deep8 = countWhere(h8, function (g) { return g.bust < 1.30; });
  var greens3 = countWhere(h12, function (g) { return g.bust >= 3; });
  var highs10 = countWhere(h20, function (g) { return g.bust >= 10; });

  var rs = redStreak(hist);
  var redRate20 = h20.length ? reds20 / h20.length : 0.5;

  var blockDanger =
    (h12.length >= 8 && reds12 >= 8 && deep8 >= 3) ||
    rs >= 7;

  var defense =
    currentDrawdown >= DEFENSE_DRAWDOWN_BITS ||
    recoveryDebt >= DEFENSE_DRAWDOWN_BITS ||
    lossStreak >= 3;

  var score = 0;
  if (rs >= 3) score += 2;
  if (rs >= 5) score += 1;
  if (redRate20 >= 0.60) score += 1;
  if (greens3 >= 4) score += 1;
  if (highs10 >= 2) score -= 0.5;
  if (blockDanger) score -= 2;
  if (defense) score -= 1.5;

  return {
    redStreak: rs,
    reds8: reds8,
    reds12: reds12,
    redRate20: redRate20,
    deep8: deep8,
    greens3: greens3,
    highs10: highs10,
    blockDanger: blockDanger,
    defense: defense,
    score: score
  };
}

function shouldPlay(moment) {
  var remainingIncludingThis = 101 - windowGames;
  var needMin = MIN_PLAY_TARGET - windowPlayed;

  if (windowPlayed >= PLAY_TARGET) return false;

  /*
   * V1.3: forza solo il minimo 50/100.
   * Da 50 a 65 entra solo se il momento lo permette.
   */
  if (needMin >= remainingIncludingThis) return true;

  var p = PLAY_TARGET / 100;

  if (moment.redStreak >= 3) p += 0.12;
  if (moment.redStreak >= 5) p += 0.04;

  if (moment.blockDanger) p -= 0.25;
  if (moment.defense) p -= 0.25;

  if (recoveryDebt > 0 && !moment.defense) p += 0.04;

  p = clamp(p, 0.15, 0.90);

  return Math.random() < p;
}

function chooseRegularPayout(moment) {
  var payout = 1.62;

  if (moment.redStreak >= 3) payout = 1.55;
  if (moment.redStreak >= 5) payout = 1.48;
  if (moment.redStreak >= 7) payout = 1.42;

  if (sessionPnl > 0) {
    payout += Math.min(0.30, winStreak * 0.06);
  }

  if (sessionPnl > TARGET_BITS * 6) payout += 0.08;

  if (winStreak >= 4 && !moment.blockDanger && !moment.defense) {
    payout += 0.15;
  }

  if (recoveryDebt > 0) {
    payout = Math.min(payout, 1.78);
  }

  if (moment.blockDanger) {
    payout = Math.min(payout, 1.55);
  }

  if (moment.defense) {
    payout = clamp(payout + 0.10, 1.55, 1.90);
  }

  payout += rnd(-0.03, 0.04);

  return r2(clamp(payout, 1.35, 2.50));
}

function recoveryTarget(moment) {
  if (recoveryDebt <= 0) return TARGET_BITS;

  /*
   * V1.3: recupero più lento e legato al deficit reale.
   */
  var slice = Math.min(
    moment && moment.defense ? 1.50 : 2.50,
    Math.max(0.30, recoveryDebt * 0.18)
  );

  return TARGET_BITS + slice;
}

function bankrollCapBits() {
  var balBits = satToBits(userInfo.balance);
  return Math.max(0.01, balBits * MAX_BANKROLL_PCT);
}

function regularPlan(moment) {
  var payout = chooseRegularPayout(moment);
  var desiredProfit = recoveryTarget(moment);

  var stake = desiredProfit / (payout - 1);

  if (moment.redStreak >= 4 && !moment.blockDanger && !moment.defense) {
    stake *= 1.05;
  }

  /*
   * V1.3: dopo perdite consecutive NON aumenta.
   * Riduce progressivamente l'esposizione.
   */
  if (lossStreak === 2) stake *= 0.85;
  if (lossStreak >= 3) stake *= 0.70;

  var modeCap = moment.defense ? DEFENSE_MAX_BET_BITS : MAX_BET_BITS;

  var cap = Math.min(
    modeCap,
    bankrollCapBits()
  );

  stake = clamp(stake, 0.01, cap);

  if (stake * (payout - 1) < desiredProfit) {
    var neededPayout = 1 + (desiredProfit / stake);

    payout = r2(
      clamp(
        Math.max(payout, neededPayout),
        1.35,
        moment.defense ? 2.20 : 4.50
      )
    );
  }

  var plannedProfit = stake * (payout - 1);

  var mode;
  if (moment.defense) mode = 'DEFENSE';
  else if (recoveryDebt > 0) mode = 'RECOVERY';
  else if (moment.redStreak >= 3) mode = 'NEG_BREAK';
  else mode = 'NORMAL';

  return {
    mode: mode,
    stakeBits: r2(stake),
    payout: payout,
    desiredProfit: r2(plannedProfit),
    minBetSat: 100
  };
}

function moonshotReady(moment) {
  if (reserveBits < RESERVE_TRIGGER) return false;
  if (moonCooldown > 0) return false;
  if (moment.blockDanger || moment.defense) return false;

  /* V1.3: Moonshot solo con sessione chiaramente positiva e debt azzerato. */
  if (sessionPnl < MOON_MIN_SESSION_PROFIT_BITS) return false;
  if (recoveryDebt > 0.01) return false;
  if (currentDrawdown > 5) return false;

  var chance = 0.06;

  if (moment.redStreak >= 4) chance += 0.03;
  if (winStreak >= 3) chance += 0.04;
  if (sessionPnl > TARGET_BITS * 10) chance += 0.02;

  return Math.random() < clamp(chance, 0.04, 0.15);
}

function moonshotPlan(moment) {
  var reserveStrength = clamp(
    (reserveBits - RESERVE_TRIGGER) /
    Math.max(1, RESERVE_TRIGGER * 4),
    0,
    1
  );

  var payout = MOON_MIN + (MOON_MAX - MOON_MIN) * reserveStrength;

  if (moment.redStreak >= 5) payout += rnd(20, 60);
  else payout += rnd(-10, 20);

  payout = r2(clamp(payout, MOON_MIN, MOON_MAX));

  var stake = reserveBits * 0.06;
  stake = Math.max(MOON_MIN_BET_BITS, stake);
  stake = Math.min(MOON_MAX_BET_BITS, stake);
  stake = Math.min(stake, bankrollCapBits());

  return {
    mode: 'MOONSHOT',
    stakeBits: r2(stake),
    payout: payout,
    desiredProfit: r2(stake * (payout - 1)),
    minBetSat: 100
  };
}

function placePlan(plan, moment) {
  var stakeSat = safeBetSat(
    plan.stakeBits,
    plan.minBetSat || 100
  );

  var payout = r2(Number(plan.payout));

  if (
    !isFinite(stakeSat) ||
    stakeSat < 100 ||
    Math.floor(stakeSat) !== stakeSat ||
    stakeSat % 100 !== 0
  ) {
    log('[BET ERROR] Puntata LIVE non valida: ' + stakeSat + ' sat');
    return false;
  }

  if (!isFinite(payout) || payout <= 1) {
    log('[BET ERROR] Payout non valido: ' + payout);
    return false;
  }

  if (stakeSat > userInfo.balance) {
    log('[SKIP] Saldo insufficiente. Richiesti ' + satToBits(stakeSat) + ' bits');
    return false;
  }

  var actualStakeBits = satToBits(stakeSat);

  try {
    engine.bet(stakeSat, payout);
  }
  catch (e) {
    log(
      '[BET ERROR LIVE] ' +
      (e && e.message ? e.message : String(e)) +
      ' | stakeSat=' + stakeSat +
      ' | bits=' + actualStakeBits +
      ' | payout=' + payout
    );
    return false;
  }

  lastMode = plan.mode;
  windowPlayed++;
  totalPlayed++;

  if (plan.mode === 'MOONSHOT') {
    reserveBits = Math.max(0, reserveBits - actualStakeBits);
    moonCooldown = 10 + Math.floor(Math.random() * 15);
  }

  log(
    '[BET] ' + plan.mode +
    ' | ' + actualStakeBits + ' bits @ ' + payout + 'x' +
    ' | target≈+' + r2(actualStakeBits * (payout - 1)) +
    ' | redStreak=' + moment.redStreak +
    ' | debt=' + r2(recoveryDebt) +
    ' | DD=' + r2(currentDrawdown) +
    ' | reserve=' + r2(reserveBits) +
    ' | quota=' + windowPlayed + '/' + PLAY_TARGET
  );

  return true;
}

function processLastResult() {
  var g = engine.history.first();

  if (!g || typeof g.bust !== 'number') return;
  if (g.id === lastProcessedGameId) return;

  lastProcessedGameId = g.id;

  if (moonCooldown > 0) moonCooldown--;

  if (!g.wager || g.wager <= 0) {
    updateRiskState();
    return;
  }

  var stakeBits = satToBits(g.wager);

  var won =
    typeof g.cashedAt === 'number' &&
    g.cashedAt > 1;

  if (won) {
    var profitBits = stakeBits * (g.cashedAt - 1);

    totalWon++;
    winStreak++;
    lossStreak = 0;
    sessionPnl += profitBits;

    if (lastMode === 'MOONSHOT') {
      reserveBits += stakeBits + profitBits;
    }
    else {
      reserveBits += profitBits * RESERVE_CUT;
    }

    updateRiskState();

    log(
      '[WIN] +' + r2(profitBits) +
      ' bits | P/L=' + r2(sessionPnl) +
      ' | debt=' + r2(recoveryDebt) +
      ' | DD=' + r2(currentDrawdown) +
      ' | reserve=' + r2(reserveBits)
    );
  }
  else {
    totalLost++;
    lossStreak++;
    winStreak = 0;
    sessionPnl -= stakeBits;

    updateRiskState();

    log(
      '[LOSS] -' + r2(stakeBits) +
      ' bits | debt=' + r2(recoveryDebt) +
      ' | P/L=' + r2(sessionPnl) +
      ' | DD=' + r2(currentDrawdown) +
      ' | reserve=' + r2(reserveBits)
    );
  }
}

engine.on('GAME_ENDED', function () {
  processLastResult();
});

engine.on('GAME_STARTING', function () {
  updateRiskState();

  var moment = analyzeMoment();

  windowGames++;

  if (windowGames > 100) {
    log(
      '[100 GAME]' +
      ' giocati=' + windowPlayed + '/100' +
      ' | W=' + totalWon +
      ' | L=' + totalLost +
      ' | P/L=' + r2(sessionPnl) +
      ' | debt=' + r2(recoveryDebt) +
      ' | DD=' + r2(currentDrawdown) +
      ' | reserve=' + r2(reserveBits)
    );

    windowGames = 1;
    windowPlayed = 0;
  }

  if (!shouldPlay(moment)) {
    log(
      '[SKIP] quota ' + windowPlayed + '/' + PLAY_TARGET +
      ' | redStreak=' + moment.redStreak +
      (moment.blockDanger ? ' | BLOCK GUARD' : '') +
      (moment.defense ? ' | DEFENSE' : '') +
      ' | DD=' + r2(currentDrawdown)
    );
    return;
  }

  var plan = moonshotReady(moment)
    ? moonshotPlan(moment)
    : regularPlan(moment);

  placePlan(plan, moment);
});

log('KIBER NEGATIVE BREAKER V1.3 LIVE avviato');
log(
  'Target: ' + MIN_PLAY_TARGET + '-' + PLAY_TARGET + '/100' +
  ' | profit target: ' + TARGET_BITS + ' bits' +
  ' | DEFENSE DD: ' + DEFENSE_DRAWDOWN_BITS + ' bits' +
  ' | Moonshot: ' + MOON_MIN + 'x-' + MOON_MAX + 'x' +
  ' | LIVE BET: bit interi'
);
