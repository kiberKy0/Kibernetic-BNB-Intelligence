engine.on("GAME_ENDED", function () {
    if (stopped) return;

    var game = engine.history.first();
    if (!game || typeof game.bust !== "number") return;

    totalRounds++;

    history.push(game.bust);
    if (history.length > MEMORY_MAX) history.shift();
    updateFiveXIntelligence(game.bust);

    if (trackingThisRound) {
        decisionWindow.push(playedThisRound ? 1 : 0);
        if (decisionWindow.length > 100) decisionWindow.shift();
    }

    if (!playedThisRound || !game.wager) {
        roundsSinceBet++;
        return;
    }

    roundsSinceBet = 0;
    totalBets++;

    var won = !!game.cashedAt;
    var wager = game.wager;
    var pnl = won ? wager * (game.cashedAt - 1) : -wager;

    addModePnl(lastMode, pnl);
    phasePnl[lastPhase] += pnl;
    recordOutcome(won, pnl, lastMode, lastPhase, lastPayout);
    if (lastRecoveryPlanBet) {
        recoveryPlanBets++;
        pnlRecoveryPlan += pnl;
    }

    if (won) {
        wins++;
        consecutiveLosses = 0;
        logResult("WIN", game, pnl);

        recoveryStep = 0;
        recoveryOriginPhase = null;
        recoveryWaits = 0;
        closeCycle();
    } else {
        losses++;
        consecutiveLosses++;
        logResult("LOSS", game, pnl);

        if (lastMode === "BASE" || lastMode === "ATTACK") {
            recoveryStep = 1;
            recoveryOriginPhase = lastPhase;
            recoveryWaits = 0;
        } else if (lastMode === "RECOVERY_1") {
            recoveryStep = 2;
            recoveryWaits = 0;
        } else if (lastMode === "RECOVERY_2") {
            recoveryStep = 0;
            recoveryOriginPhase = null;
            recoveryWaits = 0;
            cooldown = Math.max(cooldown, 8);
            closeCycle();
        }

        if (consecutiveLosses >= 2 && recoveryStep === 0) {
            cooldown = Math.max(cooldown, 5);
        }
    }

    updateBestBalance(userInfo.balance);
    updateProtection(userInfo.balance);
    updateGivebackBrake(userInfo.balance);
    updateRecoveryPlanState(userInfo.balance);

    if (totalRounds % 20 === 0) printStats();
});

function metrics(windowSize) {
    var n = Math.min(windowSize, history.length);
    if (!n) {
        return { n: 0, avg: 0, cappedAvg: 0, median: 0, p140: 0, p200: 0, p300: 0, p500: 0, max: 0 };
    }

    var arr = history.slice(history.length - n);
    var sorted = arr.slice().sort(function (a, b) { return a - b; });
    var sum = 0;
    var cappedSum = 0;
    var ge140 = 0, ge200 = 0, ge300 = 0, ge500 = 0, max = 0;

    for (var i = 0; i < arr.length; i++) {
        var v = arr[i];
        sum += v;
        cappedSum += Math.min(v, 10);
        if (v >= 1.40) ge140++;
        if (v >= 2.00) ge200++;
        if (v >= 3.00) ge300++;
        if (v >= 5.00) ge500++;
        if (v > max) max = v;
    }

    var median;
    if (n % 2 === 1) {
        median = sorted[(n - 1) / 2];
    } else {
        median = (sorted[n / 2 - 1] + sorted[n / 2]) / 2;
    }

    return {
        n: n,
        avg: sum / n,
        cappedAvg: cappedSum / n,
        median: median,
        p140: ge140 / n,
        p200: ge200 / n,
        p300: ge300 / n,
        p500: ge500 / n,
        max: max
    };
}

/* ==========================================================
   PHASE ENGINE
   Main rule requested by user:
   NEGATIVE < 2.00 avg14
   POSITIVE 2.00 - 3.49
   HOT >= 3.50
   with median/volatility confirmation
   ========================================================== */
function detectPhase(s) {
    var avg14 = s.m14.cappedAvg;
    var med14 = s.m14.median;

    if (avg14 < 2.00) return "NEGATIVE";

    if (avg14 >= 3.50) {
        /* Avoid fake HOT caused by one absurd spike */
        var hotSupport = 0;
        if (med14 >= 1.75) hotSupport++;
        if (s.m14.p200 >= 0.35) hotSupport++;
        if (s.m14.p300 >= 0.18) hotSupport++;
        if (s.m5.cappedAvg >= 2.20) hotSupport++;

        if (hotSupport >= 2) return "HOT";
        return "POSITIVE";
    }

    return "POSITIVE";
}

/* ==========================================================
   PHASE HYSTERESIS / HOT PERSISTENCE
   HOT must be confirmed twice before becoming operational.
   Leaving HOT also needs 2 POSITIVE observations, while
   NEGATIVE remains immediate for safety.
   ========================================================== */
function resolvePhaseWithPersistence(candidate) {
    hotTransitionCaution = false;

    if (candidate === "NEGATIVE") {
        hotCandidateStreak = 0;
        hotExitStreak = 0;
        return "NEGATIVE";
    }

    if (currentPhase === "HOT") {
        if (candidate === "HOT") {
            hotExitStreak = 0;
            hotCandidateStreak = 0;
            return "HOT";
        }

        hotExitStreak++;
        hotTransitionCaution = true;

        if (hotExitStreak >= HOT_EXIT_CONFIRM_ROUNDS) {
            hotExitStreak = 0;
            hotCandidateStreak = 0;
            return "POSITIVE";
        }

        return "HOT";
    }

    if (candidate === "HOT") {
        hotCandidateStreak++;
        hotExitStreak = 0;

        if (hotCandidateStreak >= HOT_CONFIRM_ROUNDS) {
            hotCandidateStreak = 0;
            hotPhaseBetCount = 0;
            return "HOT";
        }

        return "POSITIVE";
    }

    hotCandidateStreak = 0;
    hotExitStreak = 0;
    return "POSITIVE";
}

/* ==========================================================
   BREAK WATCH - NO AUTO BET
   ========================================================== */
function updateBreakWatch(s) {
    var info = detectNegativeBreak();

    if (info.detected) {
        breakWatchRounds = 3;
        breakWatchStrength = info.strength;
        lastBreakMultiplier = info.multiplier;
        log("[BREAK WATCH] x" + info.multiplier.toFixed(2), "| block", info.blockLength, "| strength", info.strength.toFixed(0));
        return;
    }

    if (breakWatchRounds > 0) {
        breakWatchRounds--;
        if (breakWatchRounds === 0) {
            breakWatchStrength = 0;
            lastBreakMultiplier = 0;
        }
    }
}

function detectNegativeBreak() {
    var n = history.length;
    if (n < 4) return { detected: false, multiplier: 0, blockLength: 0, strength: 0 };

    var last = history[n - 1];
    if (last < 1.40) return { detected: false, multiplier: last, blockLength: 0, strength: 0 };

    var count = 0;
    var sum = 0;

    for (var i = n - 2; i >= 0; i--) {
        if (history[i] < 1.40) {
            count++;
            sum += history[i];
        } else {
            break;
        }
    }

    if (count < 3) return { detected: false, multiplier: last, blockLength: count, strength: 0 };

    var avgBlock = sum / count;
    var strength = 35;
    strength += clamp((count - 3) * 8, 0, 24);
    strength += clamp((last - 1.40) * 12, 0, 30);
    strength += clamp((1.40 - avgBlock) * 30, 0, 11);

    return {
        detected: true,
        multiplier: last,
        blockLength: count,
        strength: clamp(strength, 0, 100)
    };
}

/* ==========================================================
   VOLATILITY PENALTY
   Light penalty, never a hard guard
   ========================================================== */
function computeVolatilityPenalty(s) {
    var penalty = 0;
    var avg = s.m14.cappedAvg;
    var med = s.m14.median;

    if (avg > med * 1.8) penalty += 12;
    if (avg > med * 2.4) penalty += 12;
    if (s.m14.max >= 20 && s.m14.p300 < 0.22) penalty += 12;
    if (s.m5.max >= 15 && s.m5.p200 < 0.40) penalty += 8;

    /* alternanza estrema tipo 1.05 -> 8x -> 1.10 -> 6x */
    if (s.alternation8 >= 0.55) penalty += 16;
    if (s.alternation8 >= 0.75) penalty += 14;
    if (s.alternation14 >= 0.45) penalty += 8;

    return clamp(penalty, 0, 55);
}

/* ==========================================================
   CONFIDENCE / PERSISTENCE 0-100
   ========================================================== */
function computeConfidence(s, volatilityPenalty) {
    var c = 45;

    c += clamp((phaseAge - 1) * 5, 0, 20);

    if (currentPhase === "NEGATIVE") {
        if (s.m14.cappedAvg < 1.80) c += 8;
        if (s.m14.median < 1.50) c += 7;
        if (s.lowStreak >= 3) c += 8;
    }

    if (currentPhase === "POSITIVE") {
        if (s.m14.median >= 1.60) c += 8;
        if (s.m14.p200 >= 0.32) c += 8;
        if (s.m5.p140 >= 0.60) c += 6;
    }

    if (currentPhase === "HOT") {
        if (s.m14.median >= 2.00) c += 10;
        if (s.m14.p300 >= 0.22) c += 10;
        if (s.m5.p200 >= 0.50) c += 8;
    }

    if (breakWatchRounds > 0) {
        c += breakWatchStrength * 0.12;
    }

    c -= volatilityPenalty;

    /* Self-calibration: recent bad decisions reduce effective confidence. */
    lastFeedback = computeOutcomeFeedback();
    c -= lastFeedback.basePenalty;

    return clamp(c, 0, 100);
}

/* ==========================================================
   OUTCOME FEEDBACK / SELF-CALIBRATION
   This does not predict future crashes. It only measures how
   well the bot's own recent decisions have been performing.
   ========================================================== */
function recordOutcome(won, pnl, mode, phase, payout) {
    outcomeMemory.push({
        won: !!won,
        pnl: pnl,
        mode: mode,
        phase: phase,
        payout: payout
    });

    if (outcomeMemory.length > OUTCOME_MEMORY_MAX) outcomeMemory.shift();
}

function selectRecentOutcomes(limit, phase, modePrefix) {
    var out = [];

    for (var i = outcomeMemory.length - 1; i >= 0 && out.length < limit; i--) {
        var item = outcomeMemory[i];
        if (phase && item.phase !== phase) continue;
        if (modePrefix && item.mode.indexOf(modePrefix) !== 0) continue;
        out.push(item);
    }

    return out;
}

function summarizeOutcomes(items) {
    var winsLocal = 0;
    var pnlLocal = 0;

    for (var i = 0; i < items.length; i++) {
        if (items[i].won) winsLocal++;
        pnlLocal += items[i].pnl;
    }

    return {
        n: items.length,
        wins: winsLocal,
        losses: items.length - winsLocal,
        winRate: items.length ? winsLocal / items.length : 0,
        pnl: pnlLocal
    };
}

function currentOutcomeLossStreak() {
    var streak = 0;

    for (var i = outcomeMemory.length - 1; i >= 0; i--) {
        if (!outcomeMemory[i].won) streak++;
        else break;
    }

    return streak;
}

function performancePenalty(summary, softLossRate, hardLossRate) {
    if (summary.n < 3 || summary.pnl >= 0) return 0;

    var p = 4;
    var lossRate = 1 - summary.winRate;

    if (lossRate >= softLossRate) p += 4;
    if (summary.n >= 4 && lossRate >= hardLossRate) p += 4;

    return p;
}

function computeOutcomeFeedback() {
    var globalSummary = summarizeOutcomes(selectRecentOutcomes(8, null, null));
    var phaseSummary = summarizeOutcomes(selectRecentOutcomes(6, currentPhase, null));
    var attackSummary = summarizeOutcomes(selectRecentOutcomes(5, null, "ATTACK"));
    var recoverySummary = summarizeOutcomes(selectRecentOutcomes(6, null, "RECOVERY"));
    var lossStreak = currentOutcomeLossStreak();

    var basePenalty = 0;
    basePenalty += performancePenalty(globalSummary, 0.55, 0.70);
    basePenalty += performancePenalty(phaseSummary, 0.55, 0.70);
    basePenalty += clamp((lossStreak - 1) * 3, 0, 9);
    basePenalty = clamp(basePenalty, 0, 24);

    var attackPenalty = performancePenalty(attackSummary, 0.50, 0.67);
    var recoveryPenalty = performancePenalty(recoverySummary, 0.50, 0.67);

    return {
        basePenalty: basePenalty,
        attackPenalty: clamp(attackPenalty, 0, 12),
        recoveryPenalty: clamp(recoveryPenalty, 0, 12),
        lossStreak: lossStreak
    };
}

/* ==========================================================
   TARGET POTENTIAL 0-100
   ========================================================== */
function computeTargetPotential(s, confidence, volatilityPenalty, balance) {
    var p = profitPct(balance);
    var nextTarget = p < 1 ? 1 : (p < 2 ? 2 : 5);
    var distance = Math.max(0, nextTarget - p);
    var roundsLeft = 999999; // continuous mode: no session horizon

    var tp = 35;
    tp += confidence * 0.35;

    if (currentPhase === "POSITIVE") tp += 8;
    if (currentPhase === "HOT") tp += 15;
    if (currentPhase === "NEGATIVE") tp -= 12;

    if (s.recentSlope > 0.30) tp += 8;
    if (s.recentSlope < -0.30) tp -= 8;

    if (distance <= 0.50) tp += 8;
    if (distance > 2.50) tp -= 8;

    if (roundsLeft < 30 && distance > 1.0) tp -= 10;

    if (protectionLevel === 1) tp -= 5;
    if (protectionLevel === 2) tp -= 12;

    var giveback = Math.max(0, bestProfitPct - p);
    if (protectionLevel === 1 && giveback > 0.40) tp -= 10;
    if (protectionLevel === 2 && giveback > 0.50) tp -= 15;
    if (protectionLevel === 2 && giveback > 1.00) tp -= 18;

    if (givebackBrake === 1) tp -= 8;
    if (givebackBrake === 2) tp -= 16;

    tp -= volatilityPenalty * 0.5;

    return clamp(tp, 0, 100);
}

/* ==========================================================
   PROFIT PROTECTION +1 / +2 / +5
   soft, progressive, not rigid
   ========================================================== */
function updateProtection(balance) {
    if (recoveryPlanActive) {
        protectionLevel = 0;
        return;
    }
    var p = profitPct(balance);
    if (p > bestProfitPct) bestProfitPct = p;

    if (bestProfitPct >= 2) {
        protectionLevel = 2;
    } else if (bestProfitPct >= 1) {
        protectionLevel = 1;
    } else {
        protectionLevel = 0;
    }
}

function updateGivebackBrake(balance) {
    if (recoveryPlanActive) {
        givebackBrake = 0;
        return;
    }
    var p = profitPct(balance);
    var giveback = Math.max(0, bestProfitPct - p);

    givebackBrake = 0;

    if (bestProfitPct >= 1.00 && giveback >= 0.35) {
        givebackBrake = 1;
    }

    if ((bestProfitPct >= 1.50 && giveback >= 0.55) ||
        (bestProfitPct >= 2.00 && giveback >= 0.50) ||
        (bestProfitPct >= 1.00 && giveback >= 0.85)) {
        givebackBrake = 2;
    }
}

/* ==========================================================
   PACE
   ========================================================== */
function choosePace(s, confidence, targetPotential, volatilityPenalty) {
    var score = confidence * 0.45 + targetPotential * 0.55;

    if (protectionLevel === 2) score -= 12;
    else if (protectionLevel === 1) score -= 5;

    var giveback = Math.max(0, bestProfitPct - profitPct(userInfo.balance));
    if (protectionLevel === 1 && giveback > 0.40) score -= 8;
    if (protectionLevel === 2 && giveback > 0.50) score -= 14;
    if (protectionLevel === 2 && giveback > 1.00) score -= 18;

    if (givebackBrake === 1) score -= 12;
    if (givebackBrake === 2) score -= 24;
    if (hotTransitionCaution) score -= 10;

    score -= volatilityPenalty * 0.4;

    if (currentPhase === "NEGATIVE") score -= 8;
    if (currentPhase === "HOT") score += 8;

    if (givebackBrake === 2) return "SLOW";
    if (givebackBrake === 1 && score >= 72) return "NORMAL";
    if (score >= 72) return "AGGRESSIVE";
    if (score >= 50) return "NORMAL";
    return "SLOW";
}

/* ==========================================================
   ENTRY DECISION
   ========================================================== */
function evaluateEntry(s, confidence, targetPotential, volatilityPenalty, pace, attackBlocked) {
    if (recoveryPlanActive) {
        return evaluateRecoveryPlanEntry(s, confidence, targetPotential, volatilityPenalty, pace, attackBlocked);
    }

    var threshold = 0;

    if (currentPhase === "NEGATIVE") threshold = 72;
    else if (currentPhase === "POSITIVE") threshold = 58;
    else threshold = 62;

    if (pace === "SLOW") threshold += 8;
    if (pace === "AGGRESSIVE") threshold -= 6;

    if (protectionLevel === 1) threshold += 4;
    if (protectionLevel === 2) threshold += 10;

    var giveback = Math.max(0, bestProfitPct - profitPct(userInfo.balance));
    if (protectionLevel === 1 && giveback > 0.40) threshold += 8;
    if (protectionLevel === 2 && giveback > 0.50) threshold += 14;
    if (protectionLevel === 2 && giveback > 1.00) threshold += 16;

    if (givebackBrake === 1) threshold += 6;
    if (givebackBrake === 2) threshold += 12;
    if (hotTransitionCaution) threshold += 6;

    /* Negative can play, but Break Watch only improves confidence; never auto-enters */
    if (currentPhase === "NEGATIVE" && breakWatchRounds > 0) {
        threshold -= clamp(breakWatchStrength * 0.08, 0, 6);
    }

    var decisionScore = confidence * 0.55 + targetPotential * 0.45 - volatilityPenalty * 0.25;

    if (decisionScore < threshold) {
        return { enter: false, reason: "decision " + decisionScore.toFixed(0) + " < " + threshold.toFixed(0) };
    }

    var attack = false;
    var attackConfidence = confidence - lastFeedback.attackPenalty;
    var attackMinConfidence = givebackBrake === 1 ? 82 : 72;
    var attackMinTP = givebackBrake === 1 ? 78 : 70;

    if (!attackBlocked && pace === "AGGRESSIVE" && attackConfidence >= attackMinConfidence && targetPotential >= attackMinTP) {
        if (currentPhase === "HOT" || (currentPhase === "POSITIVE" && s.m14.p200 >= 0.38)) {
            attack = true;
        }
    }

    var payout = chooseAdaptivePayout(s, confidence, targetPotential, pace, attack, false);
    var amount = chooseAdaptiveBetSize(s, confidence, targetPotential, pace, attack, false);

    return {
        enter: true,
        mode: attack ? "ATTACK" : "BASE",
        payout: payout,
        amount: amount,
        reason: "OK"
    };
}

/* ==========================================================
   ADAPTIVE PAYOUT 1.01 - 5.00
   ========================================================== */
function chooseAdaptivePayout(s, confidence, targetPotential, pace, attack, recovery) {
    if (recoveryPlanActive) {
        return chooseRecoveryPlanPayout(s, confidence, targetPotential, pace, attack, recovery);
    }

    var p;
    var med = Math.max(1.01, s.m14.median);

    if (currentPhase === "NEGATIVE") {
        p = 1.05 + confidence / 100 * 0.20;
        if (breakWatchRounds > 0) p += breakWatchStrength / 100 * 0.15;
        p = Math.min(p, Math.max(1.08, med * 0.95 + 0.10));
    } else if (currentPhase === "POSITIVE") {
        p = 1.15 + confidence / 100 * 0.35 + s.m14.p200 * 0.35 + targetPotential / 100 * 0.10;
        p = Math.min(p, Math.max(1.20, med * 0.90));
    } else {
        p = 1.35 + confidence / 100 * 0.65 + s.m14.p300 * 0.80 + targetPotential / 100 * 0.15;
        p = Math.min(p, Math.max(1.50, med * 1.05));
    }

    if (pace === "SLOW") p -= 0.10;
    if (pace === "AGGRESSIVE") p += 0.15;
    if (attack) p += 0.25;
    if (recovery) p -= 0.15;

    if (protectionLevel === 1) p -= 0.08;
    if (protectionLevel === 2) p -= 0.18;

    if (currentPhase === "HOT" && hotPhaseBetCount < 3) p -= 0.10;
    if (hotTransitionCaution) p -= 0.12;
    if (givebackBrake === 1) p -= 0.06;
    if (givebackBrake === 2) p -= 0.12;

    return roundMultiplier(clamp(p, MIN_PAYOUT, MAX_PAYOUT));
}

/* ==========================================================
   ADAPTIVE BET SIZE
   Based on fraction of 2% cycle budget
   ========================================================== */
function chooseAdaptiveBetSize(s, confidence, targetPotential, pace, attack, recovery) {
    if (recoveryPlanActive) {
        return chooseRecoveryPlanBetSize(s, confidence, targetPotential, pace, attack, recovery);
    }

    startCycleIfNeeded();

    var fraction = 0.07;

    if (currentPhase === "NEGATIVE") fraction = 0.07;
    if (currentPhase === "POSITIVE") fraction = 0.10;
    if (currentPhase === "HOT") fraction = 0.14;

    fraction += confidence / 100 * 0.06;
    fraction += targetPotential / 100 * 0.05;

    if (pace === "SLOW") fraction *= 0.65;
    if (pace === "AGGRESSIVE") fraction *= 1.15;
    if (attack) fraction *= 1.30;
    if (recovery) fraction *= recoveryStep === 2 ? 1.10 : 1.02;

    if (protectionLevel === 1) fraction *= 0.78;
    if (protectionLevel === 2) fraction *= 0.50;

    /* First 3 HOT bets are deliberately lighter. */
    if (currentPhase === "HOT" && hotPhaseBetCount < 3) fraction *= 0.72;
    if (hotTransitionCaution) fraction *= 0.75;

    if (attack && lastFeedback.attackPenalty > 0) fraction *= 0.85;
    if (recovery && lastFeedback.recoveryPenalty > 0) fraction *= 0.85;

    if (givebackBrake === 1) fraction *= 0.80;
    if (givebackBrake === 2) fraction *= 0.60;

    /* Preserve the 2.0 sizing at >=1000 bits, but automatically soften it
       for smaller balances such as 729 bits. */
    fraction *= capitalRiskScale(userInfo.balance);
    fraction = clamp(fraction, 0.03, 0.30);

    var remaining = Math.max(0, cycleBudget - cycleExposure);
    var amount = Math.floor(cycleBudget * fraction);

    amount = Math.min(amount, remaining);
    amount = Math.max(unit, amount);

    return roundSats(amount);
}

/* ==========================================================
   RECOVERY REVALIDATION
   Can continue, reduce, wait, or abort
   ========================================================== */
function evaluateRecovery(s, confidence, targetPotential, volatilityPenalty, pace) {
    var giveback = Math.max(0, bestProfitPct - profitPct(userInfo.balance));

    if (volatilityPenalty >= 30 && confidence < 72) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "volatilità alta + confidence insufficiente" };
    }

    if (recoveryOriginPhase === "HOT" && currentPhase === "NEGATIVE") {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "fase degradata HOT -> NEGATIVE" };
    }

    if (recoveryOriginPhase === "POSITIVE" && currentPhase === "NEGATIVE") {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "fase degradata POSITIVE -> NEGATIVE" };
    }

    if (currentPhase === "NEGATIVE" && breakWatchRounds === 0 && confidence < 72) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "NEGATIVE senza Break Watch forte" };
    }

    if (protectionLevel === 2 && (targetPotential < 72 || giveback > 0.75)) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "protezione +2% / profit giveback" };
    }

    var recoveryConfidence = confidence - lastFeedback.recoveryPenalty;
    var minConf = recoveryStep === 1 ? 62 : 72;
    var minTP = recoveryStep === 1 ? 56 : 66;

    if (recoveryPlanActive) {
        var rpRecoverySignal = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
        var rpRecoveryMin = recoveryStep === 1 ? 70 : 78;
        minConf += 4;
        minTP += 4;
        if (rpRecoverySignal < rpRecoveryMin) {
            recoveryWaits++;
            if (recoveryWaits >= 2) {
                recoveryWaits = 0;
                return { action: "ABORT", reason: "Recovery Plan: segnale recovery non abbastanza forte" };
            }
            return { action: "WAIT", reason: "Recovery Plan: attende qualità recovery " + recoveryWaits + "/2" };
        }
    }

    /* Early HOT R2 requires materially stronger confirmation. */
    if (recoveryStep === 2 && recoveryOriginPhase === "HOT" && hotPhaseBetCount < 3) {
        minConf = Math.max(minConf, 88);
        minTP = Math.max(minTP, 82);
    }

    if (givebackBrake === 1) {
        minConf += 4;
        minTP += 4;
    }

    if (givebackBrake === 2 && recoveryStep === 2) {
        recoveryWaits = 0;
        return { action: "ABORT", reason: "profit giveback brake: R2 bloccato" };
    }

    if (recoveryConfidence < minConf || targetPotential < minTP) {
        recoveryWaits++;
        if (recoveryWaits >= 2) {
            recoveryWaits = 0;
            return { action: "ABORT", reason: "revalidation fallita per 2 round" };
        }
        return { action: "WAIT", reason: "attende conferma recovery " + recoveryWaits + "/2" };
    }

    recoveryWaits = 0;

    var payout = chooseAdaptivePayout(s, confidence, targetPotential, pace, false, true);
    var amount = chooseAdaptiveBetSize(s, confidence, targetPotential, pace, false, true);

    if (recoveryStep === 2) amount = roundSats(amount * 1.12);

    return { action: "BET", amount: amount, payout: payout, reason: "revalidated" };
}

/* ==========================================================
   CYCLE RISK 2%
   ========================================================== */
function startCycleIfNeeded() {
    if (cycleActive) return;

    cycleActive = true;
    cycleExposure = 0;
    if (recoveryPlanActive) {
        /* Recovery is not constrained by NORMAL 2% cycle budget.
           Individual size is selected by Recovery intelligence. */
        cycleBudget = Math.max(unit, userInfo.balance);
    } else {
        cycleBudget = Math.floor(userInfo.balance * (config.phaseBudgetPct.value / 100));
        cycleBudget = Math.max(unit, cycleBudget);
    }
}

function canUseCycleBudget(amount) {
    startCycleIfNeeded();
    if (recoveryPlanActive) return amount > 0 && amount <= userInfo.balance;
    return cycleExposure + amount <= cycleBudget;
}

function closeCycle() {
    cycleActive = false;
    cycleBudget = 0;
    cycleExposure = 0;
}

/* ==========================================================
   PLACE BET
   ========================================================== */
function placeBet(amount, payout, mode, s, confidence, targetPotential, pace) {
    // LIVE FIX: Bustabit live rejects fractional-bit wagers in some cases.
    // Force every wager to a whole bit = multiple of 100 satoshis.
    amount = roundSats(amount);
    payout = roundMultiplier(payout);

    if (!isFinite(amount) || amount < 100 || amount % 100 !== 0) {
        log("[SAFETY SKIP] importo puntata non valido:", amount, "| script resta attivo");
        return;
    }

    engine.bet(amount, payout);

    playedThisRound = true;
    lastRecoveryPlanBet = recoveryPlanActive;
    lastPhase = currentPhase;
    lastPayout = payout;
    lastAmount = amount;

    startCycleIfNeeded();
    cycleExposure += amount;

    if (mode === "BASE") baseBets++;
    else if (mode === "ATTACK") attackBets++;
    else recoveryBets++;

    phaseBets[currentPhase]++;
    if (currentPhase === "HOT") hotPhaseBetCount++;

    log(
        "[BET]",
        currentPhase,
        "|", mode,
        "|", bits(amount), "bits @", payout.toFixed(2) + "x",
        "| AVG14", s.m14.cappedAvg.toFixed(2),
        "| MED14", s.m14.median.toFixed(2),
        "| CONF", confidence.toFixed(0),
        "| TP", targetPotential.toFixed(0),
        "| FB-", lastFeedback.basePenalty.toFixed(0),
        "| GB", givebackBrake,
        "|", pace,
        "| RP", recoveryPlanActive ? ("ON Q" + lastRecoveryPlanSignal.toFixed(0)) : "OFF",
        "| 5X", recoveryPlanActive ? lastFiveXContext.coherence.toFixed(0) : "-",
        "| Cycle", bits(cycleExposure) + "/" + bits(cycleBudget)
    );
}

/* ==========================================================
   BET SAFETY
   ========================================================== */
function canBet(amount) {
    if (amount <= 0 || !isFinite(amount)) {
        log("[SAFETY SKIP] puntata non valida | script resta attivo");
        return false;
    }

    if (userInfo.balance < amount || userInfo.balance < unit) {
        log("[CAPITAL WAIT] saldo insufficiente per la puntata | Balance", bits(userInfo.balance), "| Need", bits(amount), "| script continua a osservare");
        return false;
    }

    return true;
}

/* ==========================================================
   FREQUENCY
   ========================================================== */
function recentBetCount() {
    var sum = 0;
    for (var i = 0; i < decisionWindow.length; i++) sum += decisionWindow[i];
    return sum;
}

function currentBetRate() {
    if (!decisionWindow.length) return 0;
    return recentBetCount() / decisionWindow.length;
}

/* ==========================================================
   SEQUENCE HELPERS
   ========================================================== */
function currentLowStreak() {
    var count = 0;
    for (var i = history.length - 1; i >= 0; i--) {
        if (history[i] < 1.40) count++;
        else break;
    }
    return count;
}

function currentHighStreak() {
    var count = 0;
    for (var i = history.length - 1; i >= 0; i--) {
        if (history[i] >= 2.00) count++;
        else break;
    }
    return count;
}

function averageSlope(shortWindow, longWindow) {
    var a = metrics(shortWindow).cappedAvg;
    var b = metrics(longWindow).cappedAvg;
    return a - b;
}

function extremeAlternation(windowSize) {
    var n = Math.min(windowSize, history.length);
    if (n < 3) return 0;

    var arr = history.slice(history.length - n);
    var transitions = 0;
    var eligible = 0;

    for (var i = 1; i < arr.length; i++) {
        var aLow = arr[i - 1] < 1.40;
        var bLow = arr[i] < 1.40;
        var aHigh = arr[i - 1] >= 3.00;
        var bHigh = arr[i] >= 3.00;

        if (aLow || aHigh || bLow || bHigh) {
            eligible++;
            if ((aLow && bHigh) || (aHigh && bLow)) transitions++;
        }
    }

    return eligible ? transitions / eligible : 0;
}

/* ==========================================================
   GLOBAL RECOVERY PLAN
   Trigger: dynamic around 3.0%-4.5% drawdown from the starting credit.
   Goal: restore startBalance, then return to frozen NORMAL mode.
   Normal 35/100 cap is suspended while Recovery is ACTIVE.
   NORMAL hard-stop logic is suspended while Recovery is ACTIVE.
   Recovery ends only after restoring the reference credit. If the platform
   balance cannot fund a wager, the bot keeps observing instead of stopping.
   No RNG: variability comes from the existing Strider sensors.
   ========================================================== */
function normalSessionRounds() {
    return Math.max(0, totalRounds - normalSessionRoundOrigin);
}

/* Dynamic Recovery trigger. It is based on percentage drawdown, not on losing
   1-2-3 bits. Weak/deteriorating structure can bring entry closer to 3.0%;
   strong/stable structure can delay it toward 4.5%. Global Recovery then
   remains active until the original reference balance is fully restored. */
function dynamicRecoveryTriggerPct() {
    if (history.length < WARMUP) return RECOVERY_PLAN_TRIGGER_PCT;

    var m5 = metrics(5);
    var m14 = metrics(14);
    var slope = averageSlope(5, 14);
    var trigger = RECOVERY_PLAN_TRIGGER_PCT;

    if (m14.cappedAvg < 1.85) trigger -= 0.35;
    if (m5.cappedAvg < 1.60) trigger -= 0.20;
    if (slope < -0.35) trigger -= 0.20;
    if (consecutiveLosses >= 2) trigger -= 0.15;

    if (currentPhase === "HOT" && m14.median >= 1.90) trigger += 0.30;
    else if (currentPhase === "POSITIVE" && m14.median >= 1.75 && slope > 0.20) trigger += 0.20;

    return clamp(trigger, 3.00, 4.50);
}

function updateRecoveryPlanState(balance) {
    var dd = drawdownPctFromStart(balance);
    var dynamicTrigger = dynamicRecoveryTriggerPct();

    if (!recoveryPlanActive && dd >= dynamicTrigger) {
        recoveryPlanActive = true;
        recoveryPlanTargetBalance = startBalance;
        recoveryPlanStartBalance = balance;
        recoveryPlanStartRound = totalRounds;
        recoveryPlanActivations++;
        protectionLevel = 0;
        givebackBrake = 0;
        closeCycle();
        recoveryStep = 0;
        recoveryOriginPhase = null;
        recoveryWaits = 0;
        cooldown = 0;
        log(
            "[RECOVERY PLAN ON]",
            "DD", dd.toFixed(2) + "%",
            "| Dynamic trigger", dynamicTrigger.toFixed(2) + "%",
            "| Balance", bits(balance),
            "| Target", bits(recoveryPlanTargetBalance),
            "| Frequency 35/100: SUSPENDED",
            "| Session cap: NONE",
            "| NORMAL 2%/5x limits: SUSPENDED"
        );
        return;
    }

    if (recoveryPlanActive && balance >= recoveryPlanTargetBalance) {
        recoveryPlanActive = false;
        recoveryPlanRecovered++;
        normalSessionRoundOrigin = totalRounds;
        decisionWindow = [];
        protectionLevel = 0;
        givebackBrake = 0;
        bestBalance = balance;
        bestProfitPct = Math.max(0, profitPct(balance));
        closeCycle();
        recoveryStep = 0;
        recoveryOriginPhase = null;
        recoveryWaits = 0;
        cooldown = 0;
        attackLock = 0;
        roundsSinceBet = 99;
        log(
            "[RECOVERY PLAN OFF]",
            "credito ripristinato",
            "| Balance", bits(balance),
            "| Recovery rounds", totalRounds - recoveryPlanStartRound,
            "| NORMAL mode restored"
        );
    }
}

function updateFiveXIntelligence(bust) {
    /* Complete POST3 windows for previous >=5x events. */
    for (var i = 0; i < fiveXEvents.length; i++) {
        var ev = fiveXEvents[i];
        if (ev.round < totalRounds && ev.post.length < FIVE_X_CONTEXT_WINDOW) {
            ev.post.push(bust);
            if (recoveryPlanActive && ev.post.length === FIVE_X_CONTEXT_WINDOW) {
                log("[5X POST]", "event r" + ev.round, "x" + ev.value.toFixed(2), "| post", formatSequence(ev.post));
            }
        }
    }

    if (bust < FIVE_X_THRESHOLD) return;

    var preStart = Math.max(0, history.length - 1 - FIVE_X_CONTEXT_WINDOW);
    var pre = history.slice(preStart, history.length - 1);
    var previous = fiveXEvents.length ? fiveXEvents[fiveXEvents.length - 1] : null;
    var interval = previous ? totalRounds - previous.round : 0;

    fiveXEvents.push({
        round: totalRounds,
        value: bust,
        interval: interval,
        pre: pre,
        post: []
    });

    if (fiveXEvents.length > FIVE_X_MEMORY_MAX) fiveXEvents.shift();

    if (recoveryPlanActive) {
        log("[5X EVENT]", "x" + bust.toFixed(2), "| interval", interval || "first", "| pre", formatSequence(pre));
    }
}

function computeFiveXContext() {
    var events = fiveXEvents.slice();
    var intervals = [];
    var i;

    for (i = Math.max(0, events.length - 25); i < events.length; i++) {
        if (events[i].interval > 0) intervals.push(events[i].interval);
    }

    var lastEvent = events.length ? events[events.length - 1] : null;
    var gap = lastEvent ? Math.max(0, totalRounds - lastEvent.round) : 0;
    var avgInterval = averageNumbers(intervals);
    var medianInterval = medianNumbers(intervals);
    var timing = computeFiveXTimingScore(gap, intervals, medianInterval);

    var currentSeq = history.slice(Math.max(0, history.length - FIVE_X_CONTEXT_WINDOW));
    var preScores = [];
    var postScores = [];

    for (i = Math.max(0, events.length - 25); i < events.length; i++) {
        if (events[i].pre && events[i].pre.length >= 2) {
            preScores.push(sequenceSimilarity5X(currentSeq, events[i].pre));
        }
        if (events[i].post && events[i].post.length >= 2) {
            postScores.push(sequenceSimilarity5X(currentSeq, events[i].post));
        }
    }

    var structure = topAverage(preScores, 5, 50);
    var postStructure = topAverage(postScores, 5, 50);
    var sampleConfidence = clamp(events.length * 5, 0, 100);
    var postZone = !!lastEvent && gap <= FIVE_X_CONTEXT_WINDOW;

    var coherence = timing * 0.45 + structure * 0.45 + sampleConfidence * 0.10;
    if (postZone) coherence -= 14;
    if (postStructure > structure + 10) coherence -= 5;
    coherence = clamp(coherence, 0, 100);

    var expectedRemaining = medianInterval > 0 ? Math.max(0, medianInterval - gap) : 0;

    lastFiveXContext = {
        events: events.length,
        intervals: intervals.length,
        gap: gap,
        avgInterval: avgInterval,
        medianInterval: medianInterval,
        expectedRemaining: expectedRemaining,
        timing: timing,
        structure: structure,
        postStructure: postStructure,
        coherence: coherence,
        postZone: postZone
    };

    return lastFiveXContext;
}

function computeFiveXTimingScore(gap, intervals, medianInterval) {
    if (!intervals.length) return 50;
    if (intervals.length < 3) return 52;

    var scale = Math.max(2, medianInterval * 0.35);
    var weighted = 0;
    var weights = 0;

    for (var i = intervals.length - 1, age = 0; i >= 0; i--, age++) {
        var w = Math.max(0.35, 1 - age * 0.05);
        var d = Math.abs(gap - intervals[i]);
        var similarity = Math.exp(-d / scale);
        weighted += similarity * w;
        weights += w;
    }

    var density = weights ? weighted / weights : 0;
    return clamp(28 + density * 68, 20, 96);
}

function sequenceSimilarity5X(a, b) {
    var n = Math.min(a.length, b.length, FIVE_X_CONTEXT_WINDOW);
    if (n < 2) return 50;

    var score = 0;
    var weights = 0;
    for (var i = 0; i < n; i++) {
        var av = a[a.length - n + i];
        var bv = b[b.length - n + i];
        var bandA = fiveXBand(av);
        var bandB = fiveXBand(bv);
        var bandScore = bandA === bandB ? 100 : (Math.abs(bandA - bandB) === 1 ? 68 : 28);
        var logDistance = Math.abs(Math.log(Math.max(1.01, av)) - Math.log(Math.max(1.01, bv)));
        var numericScore = clamp(100 - logDistance * 48, 0, 100);
        var positionWeight = 0.75 + i * 0.20;
        score += (bandScore * 0.55 + numericScore * 0.45) * positionWeight;
        weights += positionWeight;
    }
    return weights ? score / weights : 50;
}

function fiveXBand(v) {
    if (v < 1.40) return 0;
    if (v < 2.00) return 1;
    if (v < 3.00) return 2;
    if (v < 5.00) return 3;
    return 4;
}

function topAverage(values, topN, fallback) {
    if (!values.length) return fallback;
    var arr = values.slice().sort(function (a, b) { return b - a; });
    var n = Math.min(topN, arr.length);
    var sum = 0;
    for (var i = 0; i < n; i++) sum += arr[i];
    return sum / n;
}

function averageNumbers(values) {
    if (!values.length) return 0;
    var sum = 0;
    for (var i = 0; i < values.length; i++) sum += values[i];
    return sum / values.length;
}

function medianNumbers(values) {
    if (!values.length) return 0;
    var arr = values.slice().sort(function (a, b) { return a - b; });
    var n = arr.length;
    if (n % 2) return arr[(n - 1) / 2];
    return (arr[n / 2 - 1] + arr[n / 2]) / 2;
}

function formatSequence(values) {
    if (!values || !values.length) return "[]";
    var out = [];
    for (var i = 0; i < values.length; i++) out.push(Number(values[i]).toFixed(2));
    return "[" + out.join(",") + "]";
}

function computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty) {
    var score = confidence * 0.42 + targetPotential * 0.28 + 18;

    if (currentPhase === "POSITIVE") {
        if (s.m14.median >= 1.65) score += 6;
        if (s.m14.p200 >= 0.35) score += 5;
        if (s.m5.p140 >= 0.60) score += 4;
        if (s.m14.median < 1.40) score -= 8;
    } else if (currentPhase === "HOT") {
        if (s.m14.median >= 2.00) score += 7;
        if (s.m14.p300 >= 0.22) score += 6;
        if (s.m5.p200 >= 0.50) score += 5;
        if (rawPhase !== "HOT") score -= 8;
    } else {
        if (breakWatchRounds > 0) score += clamp(breakWatchStrength * 0.10, 0, 10);
        else score -= 10;
        if (s.lowStreak >= 3) score += 3;
    }

    if (s.recentSlope > 0.25) score += 5;
    if (s.recentSlope < -0.25) score -= 7;
    if (hotTransitionCaution) score -= 8;

    score -= volatilityPenalty * 0.35;
    /* Recent losses matter, but Recovery must not become permanently paralysed
       by the NORMAL feedback penalty. */
    score -= lastFeedback.basePenalty * 0.15;
    score -= clamp(lastFeedback.lossStreak * 1.5, 0, 6);

    /* 5X intelligence is an additional recognition lens, never a prediction. */
    var fx = computeFiveXContext();
    if (fx.events >= 3) {
        if (fx.coherence >= 88 && fx.timing >= 75 && fx.structure >= 78) score += 12;
        else if (fx.coherence >= 80) score += 8;
        else if (fx.coherence >= 70) score += 4;
        else if (fx.coherence < 40) score -= 3;
        if (fx.postZone) score -= 4;
    }

    return clamp(score, 0, 100);
}

function chooseRecoveryPlanPayout(s, confidence, targetPotential, pace, attack, recovery) {
    var volatilityPenalty = computeVolatilityPenalty(s);
    var quality = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    var fx = lastFiveXContext;
    var p;

    if (currentPhase === "NEGATIVE") {
        p = 1.08 + quality / 100 * 0.62;
        if (breakWatchRounds > 0) p += breakWatchStrength / 100 * 0.35;
    } else if (currentPhase === "POSITIVE") {
        p = 1.15 + quality / 100 * 1.10 + s.m14.p200 * 0.45;
    } else {
        p = 1.30 + quality / 100 * 1.55 + s.m14.p300 * 0.85;
    }

    /* When interval + structure strongly agree, Recovery may deliberately
       choose targets above the NORMAL 5x ceiling. */
    if (fx.events >= 3) {
        if (fx.coherence >= 70) p += (fx.coherence - 65) / 100 * 2.5;
        if (fx.coherence >= 84 && fx.timing >= 72 && fx.structure >= 78) p += 1.10;
        if (fx.coherence >= 92 && quality >= 90) p += 1.80;
        if (fx.postZone) p -= 0.30;
    }

    if (pace === "SLOW") p -= 0.12;
    if (pace === "AGGRESSIVE") p += 0.25;
    if (attack) p += 0.45;
    if (recovery) p -= 0.12;
    if (hotTransitionCaution) p -= 0.15;

    /* No NORMAL 5x clamp here. The formula itself determines the target. */
    return roundMultiplier(Math.max(MIN_PAYOUT, p));
}

function chooseRecoveryPlanBetSize(s, confidence, targetPotential, pace, attack, recovery) {
    var balance = userInfo.balance;
    if (balance <= unit) return unit;

    var volatilityPenalty = computeVolatilityPenalty(s);
    var quality = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    var fx = lastFiveXContext;
    var fraction = 0.010;

    if (quality >= 72) fraction = 0.015;
    if (quality >= 78) fraction = 0.025;
    if (quality >= 84) fraction = 0.040;
    if (quality >= 89) fraction = 0.060;
    if (quality >= 94) fraction = 0.080;

    if (fx.events >= 3 && fx.coherence >= 78) fraction += 0.010;
    if (fx.events >= 3 && fx.coherence >= 88 && fx.structure >= 80) fraction += 0.015;

    if (attack) fraction *= 1.18;
    if (recovery) fraction *= recoveryStep === 2 ? 1.08 : 0.96;
    if (pace === "SLOW") fraction *= 0.70;
    if (pace === "AGGRESSIVE") fraction *= 1.12;
    if (volatilityPenalty >= 24) fraction *= 0.78;
    if (lastFeedback.basePenalty >= 18) fraction *= 0.82;

    var gapRatio = recoveryPlanTargetBalance > 0
        ? Math.max(0, recoveryPlanTargetBalance - balance) / recoveryPlanTargetBalance
        : 0;
    if (gapRatio >= 0.25 && quality >= 86) fraction *= 1.08;
    if (gapRatio >= 0.50 && quality >= 90) fraction *= 1.10;

    /* Capital-adaptive Recovery. Smaller balances keep the same intelligence
       and persistence, but the individual wager is scaled down. This avoids
       treating 729 bits like a 4000-bit bankroll. */
    fraction *= capitalRiskScale(balance);
    fraction = clamp(fraction, 0.004, recoveryMaxFractionForCapital(balance));

    /* Free adaptive amount in whole bits, not tied to NORMAL 2% cycle budget.
       It remains quality-scaled rather than a fixed martingale. */
    var amount = Math.floor(balance * fraction);
    amount = roundSats(amount);
    amount = Math.max(unit, amount);
    amount = Math.min(amount, roundSats(Math.max(unit, balance - unit)));
    return amount;
}

function chooseRecoveryPlanPace(s, confidence, targetPotential, volatilityPenalty) {
    var q = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    lastRecoveryPlanSignal = q;
    if (q >= 84) return "AGGRESSIVE";
    if (q >= 72) return "NORMAL";
    return "SLOW";
}

function evaluateRecoveryPlanEntry(s, confidence, targetPotential, volatilityPenalty, pace, attackBlocked) {
    var quality = computeRecoveryPlanSignal(s, confidence, targetPotential, volatilityPenalty);
    lastRecoveryPlanSignal = quality;
    var fx = lastFiveXContext;

    var threshold = currentPhase === "NEGATIVE" ? 74 : (currentPhase === "HOT" ? 68 : 66);
    if (volatilityPenalty >= 24) threshold += 3;
    if (lastFeedback.basePenalty >= 18) threshold += 2;
    if (lastFeedback.basePenalty >= 22) threshold += 1;
    if (hotTransitionCaution) threshold += 3;
    if (fx.events >= 3 && fx.coherence >= 82 && fx.timing >= 70 && fx.structure >= 76) threshold -= 2;
    if (fx.events >= 3 && fx.postZone) threshold += 2;

    if (quality < threshold) {
        return {
            enter: false,
            reason: "Recovery Plan quality " + quality.toFixed(0) + " < " + threshold.toFixed(0)
        };
    }

    var attack = false;
    var attackConfidence = confidence - lastFeedback.attackPenalty;
    if (!attackBlocked && quality >= 82 && attackConfidence >= 76 && targetPotential >= 72) {
        if (currentPhase === "HOT" || (currentPhase === "POSITIVE" && s.m14.p200 >= 0.40)) {
            attack = true;
        }
    }
    if (!attackBlocked && fx.events >= 3 && fx.coherence >= 88 && fx.structure >= 82 && quality >= 86) {
        attack = true;
    }

    var payout = chooseAdaptivePayout(s, confidence, targetPotential, pace, attack, false);
    var amount = chooseAdaptiveBetSize(s, confidence, targetPotential, pace, attack, false);

    return {
        enter: true,
        mode: attack ? "ATTACK" : "BASE",
        payout: payout,
        amount: amount,
        reason: "Recovery Plan validated Q" + quality.toFixed(0)
    };
}

/* ==========================================================
   CAPITAL ADAPTIVE HELPERS
   ========================================================== */
function balanceInBits(balance) {
    balance = Number(balance);
    if (!isFinite(balance) || balance <= 0) return 0;
    return balance / 100;
}

function capitalRiskScale(balance) {
    var b = balanceInBits(balance);
    if (b <= 0) return 0.55;
    /* At 729 bits => 0.729. At >=1000 bits => original 2.0 sizing. */
    return clamp(b / CAPITAL_REFERENCE_BITS, 0.55, 1.00);
}

function recoveryMaxFractionForCapital(balance) {
    var b = balanceInBits(balance);
    if (b < 500) return 0.022;
    if (b < 1000) return 0.028;
    if (b < 2000) return 0.040;
    if (b < 5000) return 0.060;
    return 0.080;
}

/* ==========================================================
   BALANCE / TARGETS
   ========================================================== */
function updateBestBalance(balance) {
    if (balance > bestBalance) bestBalance = balance;
    var p = profitPct(balance);
    if (p > bestProfitPct) bestProfitPct = p;
}

function profitPct(balance) {
    if (startBalance <= 0) return 0;
    return ((balance - startBalance) / startBalance) * 100;
}

function drawdownPctFromStart(balance) {
    if (startBalance <= 0) return 0;
    return Math.max(0, ((startBalance - balance) / startBalance) * 100);
}

/* ==========================================================
   STOP
   ========================================================== */
function finish(reason) {
    if (stopped) return;
    stopped = true;
    printStats();
    stop(reason);
}

/* ==========================================================
   UTILITIES
   ========================================================== */
function roundSats(value) {
    // LIVE FIX: always return whole bits (100 satoshis per bit).
    // Floor is deliberate so risk never exceeds the calculated amount.
    value = Number(value);
    if (!isFinite(value)) return 100;
    return Math.max(100, Math.floor(value / 100) * 100);
}

function roundMultiplier(value) {
    return Math.round(value * 100) / 100;
}

function bits(sats) {
    return (sats / 100).toFixed(2);
}

function signedBits(sats) {
    var value = sats / 100;
    return (value >= 0 ? "+" : "") + value.toFixed(2);
}

function pct(value) {
    return (value * 100).toFixed(1) + "%";
}

function clamp(value, min, max) {
    return Math.max(min, Math.min(max, value));
}
